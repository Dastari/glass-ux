import { useState } from "react";

import { z } from "zod";

import { CONTAINER_PRESETS, ContainerLab, SectionPage, Specimen } from "glass-ux/gallery/kit";

import { Form, FormActions, FormStack, formBehaviour, useAppForm } from "glass-ux/form";

import { ORGANISATIONS, TAGS, wait } from "glass-ux/gallery/sections/forms.fixtures";

interface Contact {
    name: string;
    /** The account’s displayed email address. */
    email: string;
    organisation: string | null;
    tags: string[];
    cc: string[];
    retention: number | null;
    escalate: boolean;
}

const contact = z.object({
    name: z.string().trim().min(1, "Enter the contact's name"),
    email: z.string().trim().min(1, "Enter an email address").email("Enter an email such as name@example.com.au"),
    organisation: z
        .string()
        .nullable()
        .refine((value) => value !== null, "Choose an organisation"),
    tags: z.array(z.string()).max(4, "Use up to 4 tags"),
    cc: z.array(z.string()),
    retention: z
        .number({ error: "Enter a number of days" })
        .nullable()
        .refine((value) => value !== null && value >= 7 && value <= 365, "Between 7 and 365 days"),
    escalate: z.boolean(),
});

const EMPTY: Contact = { name: "", email: "", organisation: null, tags: [], cc: [], retention: 30, escalate: false };

const EMAIL = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;

function ContactForm() {
    const [saved, setSaved] = useState(EMPTY);
    const form = useAppForm({
        ...formBehaviour,
        defaultValues: saved,
        validators: { onDynamic: contact },
        onSubmit: async ({ value }) => {
            await wait(900);
            setSaved(value);
        },
    });
    return (<Form form={form} aria-label="Site contact" className="w-full" data-testid="contact-form">
      <FormStack>
        <form.AppField name="name">{(field) => <field.TextField label="Name" autoComplete="name" width="medium"/>}</form.AppField>
        <form.AppField name="email">{(field) => <field.TextField label="Email" type="email" autoComplete="email"/>}</form.AppField>
        <form.AppField name="organisation">
          {(field) => <field.ComboboxField label="Organisation" options={ORGANISATIONS} placeholder="Find an organisation"/>}
        </form.AppField>
        <form.AppField name="tags">{(field) => <field.MultiSelectField label="Tags" optional options={TAGS} placeholder="Add tags"/>}</form.AppField>
        <form.AppField name="cc">
          {(field) => (<field.TagInputField label="Copy reports to" optional normalize={(value) => value.toLowerCase()} validate={(value) => (EMAIL.test(value) ? null : "Enter an email such as name@example.com.au")} placeholder="name@example.com.au"/>)}
        </form.AppField>
        <form.AppField name="retention">
          {(field) => <field.NumberField label="Keep tickets for" unit="days" min={1} max={999} width="short"/>}
        </form.AppField>
        <form.AppField name="escalate">
          {(field) => <field.SwitchRow label="Call after hours" description="For outages at this site only."/>}
        </form.AppField>
        <FormActions>
          <form.CancelButton />
          <form.SubmitButton>Save contact</form.SubmitButton>
        </FormActions>
      </FormStack>
    </Form>);
}

export default function FormSection() {
    return (<SectionPage>
      <Specimen title="Validation flow" description="One field per row in a single column (R15.2). Leave a field to validate it; once it shows an error it re-checks as you type, so the message clears the moment it is fixed. Save validates everything and focuses the first problem. Errors replace hints in place — nothing moves.">
        <ContactForm />
      </Specimen>

      <></>

      <ContainerLab description="The form column fills its box up to --form-measure; nothing reflows into columns. A whole form has no row or header size. Drag the free frame." presets={CONTAINER_PRESETS.filter((preset) => preset.id !== "row" && preset.id !== "header")}>
        <div className="h-full overflow-y-auto p-3" data-scroll-container>
          <ContactForm />
        </div>
      </ContainerLab>

      <></>

      <></>
    </SectionPage>);
}
