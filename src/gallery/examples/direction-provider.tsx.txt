import { useState } from "react";
import {
  DirectionProvider,
  localeDirection,
  useDirection,
  type TextDirection,
} from "glass-ux/locale";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { Breadcrumb } from "glass-ux/breadcrumb";
import { Button } from "glass-ux/button";
import { Menu } from "glass-ux/menu";
import { Pagination } from "glass-ux/pagination";
import { Segmented } from "glass-ux/segmented";
import { Sheet } from "glass-ux/sheet";
import { Slider } from "glass-ux/slider";
import { SplitPane } from "glass-ux/split-pane";
import { Switch } from "glass-ux/switch";
import { Tabs } from "glass-ux/tabs";
import { Tree } from "glass-ux/tree";

const ITEMS = [
  { value: "first", label: "First" },
  { value: "second", label: "Second" },
  { value: "third", label: "Third" },
];

/** A direction region and its portalled overlays. */
function DirectionDemo() {
  const inherited = useDirection();
  const [direction, setDirection] = useState<TextDirection>(inherited);
  const [page, setPage] = useState(0);
  return (
    <div className="flex min-w-0 flex-col gap-4">
      <Segmented
        aria-label="Reading direction"
        value={direction}
        onValueChange={setDirection}
        options={[
          { value: "ltr", label: "Left to right" },
          { value: localeDirection("ar"), label: "Right to left" },
        ]}
      />
      <DirectionProvider direction={direction}>
        <div dir={direction} data-testid="direction-demo" className="flex min-w-0 flex-col gap-4">
          <Breadcrumb
            items={[
              { id: "library", label: "Library", onSelect: () => {} },
              { id: "documents", label: "Documents" },
            ]}
          />
          <Tabs.Root defaultValue="first">
            <Tabs.List items={ITEMS} aria-label="Direction tabs" />
          </Tabs.Root>
          <Segmented
            options={ITEMS}
            defaultValue="first"
            collapse="none"
            aria-label="Direction segments"
          />
          <Slider defaultValue={30} aria-label="Direction level" bubble="never" />
          <Switch aria-label="Direction switch" />
          <Pagination
            label="Direction pages"
            pageIndex={page}
            pageSize={10}
            pageRows={10}
            rowCount={40}
            onPageChange={setPage}
          />
          <div className="h-44 min-w-0" data-testid="direction-split">
            <SplitPane
              collapseBelow={0}
              startMin={0}
              endMin={0}
              startLabel="Folders"
              endLabel="Details"
              start={
                <Tree
                  aria-label="Direction folders"
                  nodes={[
                    {
                      id: "folder",
                      label: "Documents",
                      children: [{ id: "draft", label: "Draft" }],
                    },
                  ]}
                />
              }
              end={<span className="type-body">Document details</span>}
            />
          </div>
          <div className="flex flex-wrap gap-2">
            <Sheet.Root side="start">
              <Sheet.Trigger render={<Button>Open start sheet</Button>} />
              <Sheet.Popup>
                <Sheet.Header title="Start sheet" />
                <Sheet.Body>Document details</Sheet.Body>
              </Sheet.Popup>
            </Sheet.Root>
            <Menu.Root>
              <Menu.Trigger render={<Button>Direction menu</Button>} />
              <Menu.Popup>
                <Menu.Submenu label="Export">
                  <Menu.Item>Document</Menu.Item>
                </Menu.Submenu>
              </Menu.Popup>
            </Menu.Root>
          </div>
        </div>
      </DirectionProvider>
    </div>
  );
}

export default function LocaleSection() {
  const [island, setIsland] = useState("en-US");
  return (
    <SectionPage>
      <Specimen
        title="Reading direction"
        description="Set the region’s dir attribute and DirectionProvider to the same direction. Menus, sheets and keyboard controls share the context."
      >
        <DirectionDemo />
      </Specimen>
    </SectionPage>
  );
}
