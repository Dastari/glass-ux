import { useRef, useState } from "react";
import { IconFilter, IconHelp, IconPlus, IconSettings } from "glass-ux/icons";
import { Badge } from "glass-ux/badge";
import { Button } from "glass-ux/button";
import { SearchField } from "glass-ux/input";
import { Panel, PanelBody, PanelHeader } from "glass-ux/surface";
import { Tour, useTourCompletion, type TourStep } from "glass-ux/tour";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";

const ORDERS = [
  { id: "WO-2041", title: "Replace conveyor belt B3", site: "North plant", state: "Open" },
  { id: "WO-2038", title: "Calibrate torque sensor", site: "Test lab", state: "Waiting" },
  { id: "WO-2035", title: "Inspect loading dock door", site: "South plant", state: "Open" },
];
const STORAGE_KEY = "gallery-demo:work-orders:v1";

/** A small work-orders panel with a tour of its controls. */
function TourDemo() {
  const [open, setOpen] = useState(false);
  const [step, setStep] = useState(0);
  const newRef = useRef<HTMLButtonElement>(null);
  const { completed, reset } = useTourCompletion(STORAGE_KEY);
  const steps: TourStep[] = [
    { title: "Work orders", body: "Where to search, add and open work orders." },
    {
      target: "#tour-demo-search",
      title: "Search",
      body: "Finds orders by number, title or site.",
    },
    {
      target: newRef,
      title: "New order",
      body: "Opens a blank work order for the selected site.",
      placement: "left",
    },
    {
      target: "#tour-demo-archive",
      title: "Archive",
      body: "This step's target is not on the page, so the tour skips it.",
    },
    {
      target: () => document.querySelector("#tour-demo-list li"),
      title: "Orders",
      body: "Open an order to assign it or change its state.",
      placement: "top",
    },
    {
      target: "#tour-demo-settings",
      title: "Panel settings",
      body: "Choose the columns and the default site.",
      placement: "bottom",
    },
  ];
  return (
    <div className="flex w-full flex-col gap-3">
      <div className="flex flex-wrap items-center gap-3">
        <Button
          variant="primary"
          icon={<IconHelp />}
          onClick={() => {
            setStep(0);
            setOpen(true);
          }}
        >
          Start the tour
        </Button>
        <Badge tone={completed ? "success" : "neutral"}>
          {completed ? "Completed" : "Not completed"}
        </Badge>
        <Button
          variant="ghost"
          disabledReason={completed ? undefined : "The tour has not been completed"}
          onClick={reset}
        >
          Forget completion
        </Button>
      </div>
      <Panel className="w-full max-w-160">
        <PanelHeader
          title="Work orders"
          level={3}
          actions={
            <>
              <Button
                id="tour-demo-settings"
                variant="ghost"
                size="sm"
                icon={<IconSettings />}
                aria-label="Panel settings"
              />
              <Button ref={newRef} variant="secondary" size="sm" icon={<IconPlus />}>
                New order
              </Button>
            </>
          }
        />
        <PanelBody>
          <div className="flex flex-col gap-3">
            <div className="flex items-center gap-2">
              <SearchField
                id="tour-demo-search"
                size="sm"
                aria-label="Search work orders"
                placeholder="Search work orders"
                className="flex-1"
              />
              <Button variant="ghost" size="sm" icon={<IconFilter />} aria-label="Filter" />
            </div>
            <ul id="tour-demo-list" className="flex flex-col gap-1">
              {ORDERS.map((order) => (
                <li key={order.id} className="flex items-center gap-3 rounded-item px-2.5 py-2">
                  <span className="type-caption text-fg-muted" data-numeric>
                    {order.id}
                  </span>
                  <span className="min-w-0 flex-1 truncate">{order.title}</span>
                  <span className="type-caption text-fg-muted">{order.site}</span>
                  <Badge tone={order.state === "Open" ? "info" : "warning"}>{order.state}</Badge>
                </li>
              ))}
            </ul>
          </div>
        </PanelBody>
      </Panel>
      <Tour
        steps={steps}
        open={open}
        onOpenChange={setOpen}
        step={step}
        onStepChange={setStep}
        storageKey={STORAGE_KEY}
        targetTimeout={300}
      />
    </div>
  );
}

export default function TourSection() {
  return (
    <SectionPage>
      <Specimen
        title="A tour of a panel"
        description="The first step has no target and shows in the middle. Each later step lights its target and anchors the card to it; the Archive step's target is missing and is skipped. ← and → step, Esc skips, and focus returns to the button."
        layout="stack"
        wide
      >
        <TourDemo />
      </Specimen>
    </SectionPage>
  );
}
