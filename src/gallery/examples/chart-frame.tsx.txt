import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { ChartFrame } from "glass-ux/charts";

export default function ChartFrameExample() {
  return (
    <SectionPage>
      <Specimen title="ChartFrame" layout="stack">
        <div className="h-90 w-full">
          <ChartFrame
            kind="column"
            label="Custom plot of weekly reservations"
            title="Weekly reservations"
          >
            {({ width, height }) => (
              <svg
                viewBox={`0 0 ${width} ${height}`}
                role="img"
                aria-label="Three weekly reservation totals"
              >
                {[40, 65, 80].map((value, i) => (
                  <rect
                    key={i}
                    x={(width * (i + 0.1)) / 3}
                    y={height * (1 - value / 100)}
                    width={width / 5}
                    height={(height * value) / 100}
                    fill="var(--chart-1)"
                  />
                ))}
              </svg>
            )}
          </ChartFrame>
        </div>
      </Specimen>
    </SectionPage>
  );
}
