import { useState } from "react";

import { SectionPage, Specimen } from "glass-ux/gallery/kit";

import { DEMO_MAP_CONFIG } from "glass-ux/gallery/fixtures/map";

import { PinMap } from "glass-ux/map";

export default function PinMapExample() {
    const [value, setValue] = useState<{
        lng: number;
        lat: number;
    } | null>({
        lng: 147.3,
        lat: -42.8,
    });
    return (<SectionPage>
      <Specimen title="Choose a sample location" layout="stack">
        <div className="h-90 w-full">
          <PinMap config={DEMO_MAP_CONFIG} value={value} onValueChange={setValue} label="Sample studio location"/>
        </div>
      </Specimen>
    </SectionPage>);
}
