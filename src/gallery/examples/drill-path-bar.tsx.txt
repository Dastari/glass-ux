import { useState } from "react";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { DrillPathBar, useHierarchy } from "glass-ux/charts";
import { Button } from "glass-ux/button";

export default function DrillPathBarExample() {
  const root = useHierarchy({
    label: "Studio",
    children: [{ label: "Rooms", children: [{ label: "Main gallery", value: 42 }] }],
  });
  const [depth, setDepth] = useState(2);
  const trail = [root, root.children![0]!, root.children![0]!.children![0]!].slice(0, depth + 1);
  return (
    <SectionPage>
      <Specimen title="Navigate a hierarchy" layout="stack">
        <DrillPathBar trail={trail} onSelect={setDepth} />
        <Button variant="ghost" onClick={() => setDepth(2)}>
          Reset path
        </Button>
      </Specimen>
    </SectionPage>
  );
}
