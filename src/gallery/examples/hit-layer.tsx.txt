import { HitLayer } from "glass-ux/charts";
import { useRef } from "react";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { ChartTooltip, TooltipBody, useHoverStore } from "glass-ux/charts";

export default function HitLayerExample() {
  const store = useHoverStore();
  const anchor = useRef<HTMLDivElement>(null);
  return (
    <SectionPage>
      <Specimen
        title="Pointer and keyboard plot navigation"
        description="Tab into the plot and use Left/Right to move between the three values."
        layout="stack"
      >
        <div ref={anchor} className="relative h-60 w-full">
          <svg
            viewBox="0 0 300 160"
            className="h-full w-full"
            role="img"
            aria-label="Three sample values"
          >
            {[40, 65, 80].map((value, i) => (
              <circle key={i} cx={50 + i * 100} cy={160 - value} r={6} fill="var(--chart-1)" />
            ))}
          </svg>
          <HitLayer
            store={store}
            count={3}
            label="Sample reservations"
            locate={(x) => Math.max(0, Math.min(2, Math.floor(x / 100)))}
            anchor={(index) => ({ x: 50 + index * 100, y: 80 })}
            valueText={(index) => `Week ${index + 1}: ${[40, 65, 80][index]} reservations`}
          />
        </div>
        <ChartTooltip anchor={anchor} store={store}>
          {(hover) => (
            <TooltipBody
              title={`Week ${hover.index + 1}`}
              rows={[
                {
                  key: "booked",
                  label: "Reservations",
                  value: String([40, 65, 80][hover.index]),
                  slot: 1,
                },
              ]}
            />
          )}
        </ChartTooltip>
      </Specimen>
    </SectionPage>
  );
}
