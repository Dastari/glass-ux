import { useState } from "react";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { EmailInput, EmailSuggestion, suggestEmail } from "glass-ux/email-field";
import { Field } from "glass-ux/field";
import { FormStack } from "glass-ux/form";

function SuggestionDemo() {
  const [value, setValue] = useState("avery@hotmial.com");
  const suggestion = suggestEmail(value);
  return (
    <Field
      label="Email"
      hint={
        suggestion ? <EmailSuggestion suggestion={suggestion} onAccept={setValue} /> : undefined
      }
    >
      <EmailInput value={value} onValueChange={setValue} />
    </Field>
  );
}

export default function EmailFieldSection() {
  return (
    <SectionPage>
      <Specimen
        title="Typo hint"
        description="Once focus leaves, a domain one or two keystrokes from a common provider — or from the organisation's own domain — offers the correction in the message line. Choosing it replaces the value; nothing moves."
      >
        <FormStack className="w-full">
          <SuggestionDemo />
        </FormStack>
      </Specimen>
    </SectionPage>
  );
}
