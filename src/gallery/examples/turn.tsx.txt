import { useContext, useState, type ReactNode } from "react";
import { DemoScope } from "glass-ux/gallery/demo-scope";
import {
  AssistantMessage,
  Composer,
  ConversationLog,
  Turn,
  UserMessage,
  type AiMessage,
  type RunState,
} from "glass-ux/ai";
import { ContainerLab, SectionPage, Specimen } from "glass-ux/gallery/kit";
import { CONVERSATION, MODEL, NOW } from "glass-ux/gallery/sections/ai.fixtures";
import { AiGallery, LabScroll } from "glass-ux/gallery/sections/ai.kit";

const turn = CONVERSATION.find((item) => item.kind === "turn" && item.index === 1);
const first = turn?.kind === "turn" ? turn : null;
const base = (extra: Partial<AiMessage>): AiMessage => ({
  id: "x",
  role: "assistant",
  createdAt: NOW - 60000,
  state: "complete",
  model: MODEL.name,
  blocks: [],
  ...extra,
});
const FOLLOW_ANSWER =
  "The sample printer has been online since **10:12 am** and its queue is empty. Two documents printed on the Level 2 printer in the last ten minutes; both completed.\n\n- The print spooler has not stopped since the restart\n- The driver is still `4.2.1`; the fix is in `4.3.0`\n\nI can update the driver tonight after 6 pm, once the studio has signed out.";
const FOLLOW_EARLIER = [1, 2, 3].map((index) => ({
  index,
  user: {
    id: `fu${index}`,
    role: "user",
    createdAt: NOW - 600000,
    state: "complete",
    blocks: [{ kind: "markdown", text: `Sample question ${index} about the studio printers?` }],
  } as AiMessage,
  reply: base({
    id: `fa${index}`,
    blocks: [
      {
        kind: "markdown",
        text: Array.from(
          { length: 3 },
          (_, p) =>
            `Sample paragraph ${p + 1}: the studio queue has ${index + p} waiting jobs and two printers online.`,
        ).join("\n\n"),
      },
    ],
  }),
}));
const pause = (ms: number) => new Promise<void>((resolve) => setTimeout(resolve, ms));

/** A log the reader follows while an answer streams in: a tool call, then words, then the footer. */
function FollowingLog() {
  const [items, setItems] = useState(FOLLOW_EARLIER);
  const [run, setRun] = useState<RunState>("idle");
  const update = (id: string, change: (message: AiMessage) => AiMessage) =>
    setItems((current) =>
      current.map((item) => (item.reply.id === id ? { ...item, reply: change(item.reply) } : item)),
    );
  const send = async (text: string) => {
    const id = `live-${items.length + 1}`;
    setRun("sending");
    await pause(250);
    setItems((current) => [
      ...current,
      {
        index: current.length + 1,
        user: {
          id: `${id}-q`,
          role: "user",
          createdAt: Date.now(),
          state: "complete",
          blocks: [{ kind: "markdown", text }],
        },
        reply: base({ id, createdAt: Date.now(), state: "streaming" }),
      },
    ]);
    setRun("running");
    // An earlier answer grows on its own meanwhile, as work beside a pending approval does.
    const grow = (ms: number, text: string) =>
      setTimeout(
        () =>
          update("fa3", (reply) => ({
            ...reply,
            blocks: [...reply.blocks, { kind: "markdown", text }],
          })),
        ms,
      );
    grow(700, "Checking the Level 2 printer's queue and toner.");
    grow(2600, "While you decide: the Level 2 printer is online and toner is at 64 %.");
    await pause(500);
    update(id, (reply) => ({
      ...reply,
      blocks: [
        {
          kind: "tools",
          calls: [
            {
              id: `${id}-t`,
              toolId: "sample.query",
              label: "Read the printer's status",
              activeLabel: "Reading the printer's status",
              status: "running",
            },
          ],
        },
      ],
    }));
    await pause(1400);
    update(id, (reply) => ({
      ...reply,
      blocks: [
        {
          kind: "tools",
          calls: [
            {
              id: `${id}-t`,
              toolId: "sample.query",
              label: "Read the printer's status",
              status: "succeeded",
              elapsedMs: 1380,
            },
          ],
        },
        { kind: "markdown", text: "" },
      ],
    }));
    let shown = "";
    for (const word of FOLLOW_ANSWER.split(/(?<=\s)/)) {
      shown += word;
      const partial = shown;
      update(id, (reply) => ({
        ...reply,
        blocks: reply.blocks.map((block) =>
          block.kind === "markdown" ? { kind: "markdown", text: partial } : block,
        ),
      }));
      await pause(28);
    }
    update(id, (reply) => ({ ...reply, state: "complete" }));
    setRun("idle");
  };
  return (
    <div className="h-120 w-full" data-testid="conversation-follow">
      <ConversationLog
        surface="panel"
        label="Sample conversation"
        footer={
          <Composer
            aria-label="Message the assistant"
            run={run}
            onSend={(text) => void send(text)}
            onStop={() => undefined}
          />
        }
      >
        {items.map((item, position) => (
          <Turn key={item.index} index={item.index}>
            <UserMessage message={item.user} />
            <AssistantMessage message={item.reply} latest={position === items.length - 1} />
          </Turn>
        ))}
      </ConversationLog>
    </div>
  );
}

function SampleTurn({ children }: { children: ReactNode }) {
  const scope = useContext(DemoScope);
  return (
    <Turn index={1} aria-label={scope ? `${scope}: Turn 1` : "Turn 1"}>
      {children}
    </Turn>
  );
}

export default function AiMessagesSection() {
  return (
    <AiGallery>
      <SectionPage>
        <Specimen
          title="A turn"
          description="The prompt is a compact bubble on the right with its attachments above. The answer has no bubble; tables, charts, images and code break out wider. Message actions show on hover and focus, and always on the latest answer."
          wide
          layout="bare"
        >
          <div className="ai-stack">
            {first ? (
              <SampleTurn>
                <UserMessage message={first.user} editable />
                {first.replies.map((message) => (
                  <AssistantMessage key={message.id} message={message} latest />
                ))}
              </SampleTurn>
            ) : null}
          </div>
        </Specimen>

        <Specimen
          title="Follow a streaming answer"
          description="A reader at the end of the log stays there while an answer streams in and completes: send a message and the tool call, each word and the answer's actions keep the end in view. Scrolled up, nothing moves and New messages appears instead."
          wide
          layout="bare"
        >
          <FollowingLog />
        </Specimen>

        <ContainerLab description="An answer in a row, a header, a tile, a pane and a tall narrow column: the column scrolls, the mark steps aside below 22rem, prose drops to the body size below 34rem.">
          {() => (
            <LabScroll>
              {first ? (
                <SampleTurn>
                  <UserMessage message={first.user} />
                  <AssistantMessage
                    message={base({
                      id: "lab",
                      blocks: [
                        {
                          kind: "markdown",
                          text: "**6 endpoints** at [Northstar Studio](/organisations/northstar) have not checked in for more than 24 hours.",
                        },
                      ],
                    })}
                    latest
                  />
                </SampleTurn>
              ) : null}
            </LabScroll>
          )}
        </ContainerLab>
      </SectionPage>
    </AiGallery>
  );
}
