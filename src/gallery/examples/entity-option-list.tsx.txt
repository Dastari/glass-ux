import { useState } from "react";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { AiGallery } from "glass-ux/gallery/sections/ai.kit";
import { E } from "glass-ux/gallery/sections/ai.fixtures";
import { EntityOptionList } from "glass-ux/ai";
import { Input } from "glass-ux/input";

export default function EntityOptionListExample() {
  const [active, setActive] = useState(0);
  const [picked, setPicked] = useState<string>();
  return (
    <AiGallery>
      <SectionPage>
        <Specimen title="Selectable entity search rows" layout="stack">
          <Input
            aria-label="Search sample entities"
            onKeyDown={(event) => {
              if (event.key === "ArrowDown") setActive((value) => Math.min(2, value + 1));
              if (event.key === "ArrowUp") setActive((value) => Math.max(0, value - 1));
              if (event.key === "Enter") setPicked([E.c106Online, E.northstar, E.mira][active]!.id);
            }}
            aria-controls="sample-entity-list"
          />
          <EntityOptionList
            id="sample-entity-list"
            options={[E.c106Online, E.northstar, E.mira]}
            active={active}
            onActive={setActive}
            selected={(entity) => entity.id === picked}
            onPick={(entity) => setPicked(entity.id)}
          />
        </Specimen>
      </SectionPage>
    </AiGallery>
  );
}
