function byOrganisation(endpoints: DemoEndpoint[]): TreeNode[] {
  return ORGANISATIONS.map((org) => ({
    id: `org:${org}`,
    label: org,
    icon: <IconOrganisation />,
    children: endpoints
      .filter((endpoint) => endpoint.organisation === org)
      .map((endpoint) => ({
        id: endpoint.id,
        label: endpoint.name,
        description: endpoint.user,
        status: endpoint.status,
        icon: <IconEndpoint />,
      })),
  }));
}

const NODES = byOrganisation(ENDPOINTS);
const ALL_OPEN = NODES.map((node) => node.id);

function SidebarTree({
  initialQuery = "",
  nodes = NODES,
  height = "h-140",
}: {
  initialQuery?: string;
  nodes?: TreeNode[];
  height?: string;
}) {
  const [query, setQuery] = useState(initialQuery);
  const [selected, setSelected] = useState<string | null>("nst-c106");
  const total = useMemo(
    () => nodes.reduce((sum, node) => sum + (node.children?.length ?? 0), 0),
    [nodes],
  );
  return (
    <div className={`flex w-72 max-w-full flex-col gap-2 ${height}`}>
      <div className="flex items-center justify-between px-2.5">
        <span className="fa-overline">Endpoints</span>
        <CountBadge value={total} />
      </div>
      <SearchField
        size="sm"
        aria-label="Search endpoints"
        placeholder="Search endpoints"
        value={query}
        onValueChange={(value) => setQuery(String(value))}
        onClear={() => setQuery("")}
      />
      <div className="min-h-0 flex-1">
        <Tree
          nodes={nodes}
          aria-label="Endpoints"
          defaultExpanded={ALL_OPEN.slice(0, 3)}
          selectedId={selected}
          onSelectedIdChange={setSelected}
          query={query}
          emptyText={(q) => `No endpoints match “${q}”`}
        />
      </div>
    </div>
  );
}

/** The endpoint's ActionSet on right-click; organisation rows have none here. */
function withActions(node: TreeNode, row: ReactElement) {
  const target = ACTION_TARGETS.find((endpoint) => endpoint.name === node.label);
  if (!target) return row;
  return (
    <ActionContextMenu set={endpointActions} targets={[target]} label={target.name}>
      {row}
    </ActionContextMenu>
  );
}

function LinkedTree() {

      <Tree

        aria-label="Endpoints (links)"
        defaultExpanded={ALL_OPEN}
        defaultSelectedId="nst-c106"
        renderLink={(node, props) => (
          <Link
            to="/demo/o/$org/devices/$id"
            params={{ org: "all", id: node.label }}
            {...props}
            activeOptions={{ exact: true }}
            activeProps={{}}
            inactiveProps={{}}
          />
        )}
        rowWrapper={withActions}
      />
    </div>
  );
}

function subkeys(id: string): TreeNode[] {
  const leaf = (key: string, label: string): TreeNode => ({
    id: `${id}/${key}`,
    label,
    hasChildren: false,
  });
  const branch = (key: string, label: string): TreeNode => ({
    id: `${id}/${key}`,
    label,
    hasChildren: true,
  });
  if (id === "studio")
    return [
      branch("fonts", "Fonts"),
      branch("colours", "Colours"),
      leaf("name", "Display name"),
      leaf("zone", "Time zone"),
    ];
  if (id === "workspace")
    return [branch("layout", "Layout"), leaf("theme", "Theme"), leaf("density", "Density")];
  if (id === "archive") return [leaf("retention", "Retention"), leaf("location", "Location")];
  if (id === "studio/fonts") return FONT_NAMES.map((name) => leaf(name.toLowerCase(), name));
  if (id === "studio/colours")
    return ["Accent", "Canvas", "Surface"].map((name) => leaf(name.toLowerCase(), name));
  if (id === "workspace/layout")
    return [leaf("drawer", "Drawer width"), leaf("inspector", "Inspector width")];
  return [];
}

const PAGE = 10;
const later = (run: () => void, fail = false) =>
  new Promise<void>((resolve, reject) =>
    setTimeout(() => {
      if (fail) return reject(new Error("The store did not answer"));
      run();
      resolve();
    }, 600),
  );

function patchNode(list: TreeNode[], id: string, update: (node: TreeNode) => TreeNode): TreeNode[] {
  return list.map((node) =>
    node.id === id
      ? update(node)
      : node.children
        ? { ...node, children: patchNode(node.children, id, update) }
        : node,
  );
}

function ExplorerTree() {
  const [nodes, setNodes] = useState(STORE_ROOTS);
  const [selected, setSelected] = useState<string | null>(null);
  const failed = useRef(new Set<string>());
  const loadChildren = (node: TreeNode) => {
    const fail = node.id === "archive" && !failed.current.has(node.id);
    if (fail) failed.current.add(node.id);
    return later(() => {
      const all = subkeys(node.id);
      setNodes((list) =>
        patchNode(list, node.id, (current) => ({
          ...current,
          children: all.slice(0, PAGE),
          hasMore: all.length > PAGE,
        })),
      );
    }, fail);
  };
  const loadMore = (node: TreeNode) =>
    later(() => {
      const all = subkeys(node.id);
      setNodes((list) =>
        patchNode(list, node.id, (current) => {
          const count = (current.children?.length ?? 0) + PAGE;
          return { ...current, children: all.slice(0, count), hasMore: all.length > count };
        }),
      );
    });
  return (
    <div className="flex w-72 max-w-full flex-col gap-2">
      <div className="h-90" data-testid="tree-explorer">
        <Tree
          nodes={nodes}
          aria-label="Settings keys"
          selectableGroups
          selectedId={selected}
          onSelectedIdChange={setSelected}
          onLoadChildren={loadChildren}
          onLoadMore={loadMore}
        />
      </div>
      <p className="type-caption text-fg-muted" aria-live="polite">
        {selected ? `Selected: ${selected.split("/").join(" › ")}` : "No key selected"}
      </p>
    </div>
  );
}
