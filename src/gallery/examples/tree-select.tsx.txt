import { useCallback, useRef, useState } from "react";
import { IconUsers } from "glass-ux/icons";
import { Field } from "glass-ux/field";
import { FormStack } from "glass-ux/form";
import { Segmented } from "glass-ux/segmented";
import { TreeSelect, type CheckStrategy } from "glass-ux/tree-select";

import { wait } from "glass-ux/gallery/sections/forms.fixtures";
import {
  DEVICE_GROUPS,
  ORG_CHART,
  deviceGroupChildren,
} from "glass-ux/gallery/sections/pickers.fixtures";

const STRATEGIES: Array<{
  value: CheckStrategy;
  label: string;
}> = [
  { value: "parent", label: "parent" },
  { value: "child", label: "child" },
  { value: "all", label: "all" },
];

function StrategyDemo() {
  const [strategy, setStrategy] = useState<CheckStrategy>("parent");
  const [value, setValue] = useState<string[]>(["platform", "p-kenji"]);
  return (
    <FormStack>
      <Field label="Check strategy" nativeLabel={false}>
        <Segmented options={STRATEGIES} value={strategy} onValueChange={setStrategy} />
      </Field>
      <Field label="Notify" nativeLabel={false} hint={`value: ${JSON.stringify(value)}`}>
        <TreeSelect
          multiple
          nodes={ORG_CHART}
          checkStrategy={strategy}
          value={value}
          onValueChange={setValue}
          placeholder="Choose teams or people"
          defaultExpanded={["kestrel", "engineering"]}
        />
      </Field>
    </FormStack>
  );
}

/** Replace a branch's children in a tree (what an application does when they arrive). */
function withChildren(nodes: TreeNode[], id: string, children: TreeNode[]): TreeNode[] {
  return nodes.map((node) =>
    node.id === id
      ? { ...node, children }
      : node.children
        ? { ...node, children: withChildren(node.children, id, children) }
        : node,
  );
}

function LazyDemo() {
  const [nodes, setNodes] = useState(DEVICE_GROUPS);
  const failed = useRef(false);
  const onLoadChildren = useCallback(async (node: TreeNode) => {
    await wait(700);
    // The first load of Test lab fails, to show Retry.
    if (node.id === "dg-lab" && !failed.current) {
      failed.current = true;
      throw new Error("Timed out");
    }
    setNodes((current) => withChildren(current, node.id, deviceGroupChildren(node.id)));
  }, []);
  return (
    <Field
      label="Apply to devices"
      nativeLabel={false}
      hint="Members load when a group opens; Test lab fails once."
    >
      <TreeSelect
        multiple
        nodes={nodes}
        onLoadChildren={onLoadChildren}
        placeholder="Choose device groups"
      />
    </Field>
  );
}
