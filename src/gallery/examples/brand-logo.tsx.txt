import { BRAND_SOURCES } from "glass-ux/gallery/fixtures/brand";

import { Example, SectionPage, Specimen } from "glass-ux/gallery/kit";

import { BrandLogo, type BrandTones } from "glass-ux/brand";

const SOURCES = Object.values(BRAND_SOURCES);

const TONES: Array<[
    BrandTones,
    string
]> = [
    ["palette", "Palette: follows the palette and mode"],
    ["brand", "Brand: Orbit Workspace's own tones in any palette"],
    ["mono", "Mono: currentColor"],
];

export default function BrandSection() {
    return (<SectionPage>


      <Specimen title="Tones" description="Palette is the default (src/app/brand.ts). Brand keeps the brand's own palette as an island on the mark; mono draws both halves in the current text colour." wide layout="stack">
        {TONES.map(([tones, label]) => (<Example key={tones} label={label}>
            <div className="gallery-brand-row" data-testid={`brand-tones-${tones}`}>
              {SOURCES.map((source) => (<BrandLogo key={source.id} source={source} tones={tones} className={source.id === "orbit" ? "gallery-wordmark" : "gallery-mark-lg"}/>))}
            </div>
          </Example>))}
      </Specimen>








    </SectionPage>);
}
