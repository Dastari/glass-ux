import { useState } from "react";
import { CONTAINER_PRESETS, ContainerLab, SectionPage, Specimen } from "glass-ux/gallery/kit";
import { DEMO_MAP_CONFIG } from "glass-ux/gallery/fixtures/map";
import { AddressInput, AddressView, emptyAddress, type AddressValue } from "glass-ux/address-field";
import { FormStack } from "glass-ux/form";
import { demoGeocoder } from "glass-ux/gallery/fixtures/geocoder";
import { HOME_BIAS } from "glass-ux/gallery/sections/inputs.kit";

const AU = { code: "AU", name: "Australia" };
const HOBART_OFFICE: AddressValue = {
  line1: "12 Example Lane",
  line2: "Level 2",
  locality: "Hobart",
  region: { code: "AU-TAS", name: "Tasmania" },
  postcode: "7000",
  country: AU,
  location: { lat: -42.88334, lng: 147.32688 },
  pinned: false,
};
const SAMPLE_DEPOT: AddressValue = {
  line1: "24 Sample Avenue",
  line2: "",
  locality: "Mossbridge",
  region: { code: "AU-TAS", name: "Tasmania" },
  postcode: "7190",
  country: AU,
  location: { lat: -42.50843, lng: 147.91032 },
  pinned: true,
};

function LookupDemo() {
  const [value, setValue] = useState<AddressValue>(() => emptyAddress(AU));
  return (
    <FormStack className="w-full" data-testid="address-lookup">
      <AddressInput
        geocoder={demoGeocoder}
        label="Site address"
        value={value}
        onValueChange={setValue}
        bias={HOME_BIAS}
        countries={["AU", "NZ"]}
        map={DEMO_MAP_CONFIG}
      />
    </FormStack>
  );
}

export default function AddressFieldSection() {
  return (
    <SectionPage>
      <Specimen
        title="Lookup"
        description="Type three characters and suggestions arrive from the geocoder, leaning towards the home region. Choosing one fills every field and drops the pin; the fields stay editable. Enter manually opens them without a lookup."
      >
        <LookupDemo />
      </Specimen>

      <Specimen
        title="Filled, with the pin"
        description="Drag the pin, click the map or focus the pin and use the arrow keys: the position becomes “placed by hand” and Reset pin returns to the lookup's point."
      >
        <FormStack className="w-full" data-testid="address-filled">
          <AddressInput
            geocoder={demoGeocoder}
            label="Site address"
            defaultValue={HOBART_OFFICE}
            bias={HOME_BIAS}
            map={DEMO_MAP_CONFIG}
          />
        </FormStack>
      </Specimen>

      <Specimen
        title="States"
        description="Errors replace the hint in place; disabled keeps the layout."
        layout="grid"
      >
        <FormStack>
          <AddressInput
            geocoder={demoGeocoder}
            label="Site address"
            defaultValue={{ ...SAMPLE_DEPOT, postcode: "719" }}
            errors={{ postcode: "Check the postcode, such as 7000", locality: undefined }}
          />
        </FormStack>
        <FormStack>
          <AddressInput
            geocoder={demoGeocoder}
            label="Site address"
            defaultValue={HOBART_OFFICE}
            disabled
          />
        </FormStack>
      </Specimen>

      <Specimen title="Over glass and imagery" backdrop>
        <FormStack>
          <AddressInput geocoder={demoGeocoder} label="Site address" defaultValue={HOBART_OFFICE} />
        </FormStack>
      </Specimen>

      <ContainerLab
        description="The editor is a form column: it fills its box to --form-measure, one row per part. The read-only view wraps nothing and truncates long lines."
        presets={CONTAINER_PRESETS.filter(
          (preset) => preset.id !== "row" && preset.id !== "header",
        )}
      >
        <div className="h-full overflow-y-auto p-3" data-scroll-container>
          <FormStack>
            <AddressView value={HOBART_OFFICE} />
            <AddressInput
              geocoder={demoGeocoder}
              label="Site address"
              defaultValue={SAMPLE_DEPOT}
            />
          </FormStack>
        </div>
      </ContainerLab>
    </SectionPage>
  );
}
