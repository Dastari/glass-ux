import { useMemo, useState } from "react";

import { SectionPage, Specimen } from "glass-ux/gallery/kit";

import { IconSearch } from "glass-ux/icons";

import { ActionProvider, usePaletteActions } from "glass-ux/actions";

import { Button } from "glass-ux/button";

import { CommandPanel, PaletteProvider, useCommandPalette, useRegisteredCommands } from "glass-ux/command-palette";

import { Kbd } from "glass-ux/kbd";

import { ENDPOINTS, endpointActions, type DemoEndpoint } from "glass-ux/gallery/sections/overlays.fixtures";

/* ── Palette ──────────────────────────────────────────────────────────────── */
function PaletteActions({ endpoint }: {
    endpoint: DemoEndpoint;
}) {
    const targets = useMemo(() => [endpoint], [endpoint]);
    usePaletteActions(endpointActions, targets, endpoint.name);
    return null;
}

function PaletteDemo() {
    const palette = useCommandPalette();
    const { commands } = useRegisteredCommands();
    return (<div className="flex w-full flex-col items-start gap-3">
      <Button icon={<IconSearch />} iconEnd={<Kbd keys="mod+k" size="sm"/>} onClick={palette.open}>
        Search
      </Button>
      <div className="glass-overlay fa-palette-frame">
        <CommandPanel commands={commands} onRun={(command) => void command.run()} defaultQuery=""/>
      </div>
    </div>);
}

/* ── Section ──────────────────────────────────────────────────────────────── */
export default function ActionsSection() {
    const [selected, setSelected] = useState("nst-c106");
    const [fail, setFail] = useState(false);
    const endpoint = ENDPOINTS.find((item) => item.id === selected) ?? ENDPOINTS[0]!;
    return (<ActionProvider>
      <PaletteProvider>
        <PaletteActions endpoint={endpoint}/>
        <SectionPage>
          <></>

          <></>

          <Specimen title="Command palette" description="While a row is selected its actions are registered in the palette under its name, submenus flattened to “Power › Restart”. Press ⌘K or Ctrl K, or use the panel." wide>
            <PaletteDemo />
          </Specimen>

          <></>

          <></>
        </SectionPage>
      </PaletteProvider>
    </ActionProvider>);
}
