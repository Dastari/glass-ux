function TicketCard({ ticket }: { ticket: Ticket }) {
  return (
    <KanbanCard
      title={ticket.title}
      meta={ticket.meta}
      badges={
        <>
          <Badge size="sm" tone={ticket.priority === "P1" ? "danger" : "neutral"}>
            {ticket.priority}
          </Badge>
          <Badge size="sm">{ticket.kind}</Badge>
        </>
      }
      avatars={
        ticket.assignees.length ? (
          <AvatarGroup size="xs" people={ticket.assignees.map((name) => ({ name }))} />
        ) : null
      }
    />
  );
}

function ServiceBoard() {
  const [tickets, setTickets] = useState(TICKETS);
  const next = useRef(2050);
  return (
    <div className="h-150 w-full">
      <Kanban
        aria-label="Service desk"
        columns={COLUMNS}
        cards={tickets}
        onCardsChange={setTickets}
        renderCard={(ticket) => <TicketCard ticket={ticket} />}
        onAddCard={(columnId) => {
          const id = `INC-${next.current++}`;
          setTickets((current) => [
            ...current,
            {
              id,
              columnId,
              title: "New request",
              meta: `${id} · Service desk`,
              priority: "P3",
              kind: "Request",
              assignees: [],
            },
          ]);
        }}
      />
    </div>
  );
}

function SavingBoard() {
  const [tickets, setTickets] = useState(TICKETS);
  return (
    <div className="h-150 w-full">
      <Kanban
        aria-label="Saving board"
        columns={COLUMNS}
        cards={tickets}
        onCardsChange={setTickets}
        renderCard={(ticket) => <TicketCard ticket={ticket} />}
        onCardMove={(move) =>
          // The service desk resolves only work that was in progress.
          new Promise<void>((resolve, reject) =>
            window.setTimeout(
              () =>
                move.to === "resolved" && move.from !== "progress"
                  ? reject(new Error("Only tickets in progress can be resolved"))
                  : resolve(),
              700,
            ),
          )
        }
      />
    </div>
  );
}

        <SavingBoard />
      </Specimen>
    </SectionPage>
  );
}
