import { useState } from "react";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { IconOrganisation } from "glass-ux/icons";
import {
  Window,
  WindowContentHost,
  WindowFrame,
  WindowManagerProvider,
  createWindowManager,
  registerWindowKind,
} from "glass-ux/windows";

function SampleWindowBody() {
  return (
    <Window.Body>
      <p>Fictional exhibition notes.</p>
    </Window.Body>
  );
}

// Kinds register once at module scope, never during render: registering notifies every
// mounted WindowContentHost.
registerWindowKind({
  kind: "docs.notes",
  icon: IconOrganisation,
  title: () => "Exhibition notes",
  load: async () => ({ default: SampleWindowBody }),
});

function makeWindowDemo() {
  const manager = createWindowManager({ storageKey: null, channelName: null });
  const id = manager.open({ kind: "docs.notes", params: {} });
  return { manager, id };
}

export default function WindowContentHostExample() {
  const [{ manager, id }] = useState(makeWindowDemo);
  return (
    <SectionPage>
      <Specimen title="Load a registered window kind" layout="stack">
        <WindowManagerProvider manager={manager}>
          <WindowFrame.Root>
            <WindowFrame.AppBar title="Exhibition notes" />
            <WindowContentHost id={id} kind="docs.notes" params={{}} />
          </WindowFrame.Root>
        </WindowManagerProvider>
      </Specimen>
    </SectionPage>
  );
}
