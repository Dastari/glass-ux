import { z } from "zod";

import { SectionPage, Specimen } from "glass-ux/gallery/kit";

import { Form, FormActions, FormStack, formBehaviour, useAppForm } from "glass-ux/form";

import { macSchema, type MacVendorLookup } from "glass-ux/mac-field";

import { Usage } from "glass-ux/gallery/sections/inputs.kit";

import { wait } from "glass-ux/gallery/sections/forms.fixtures";

/** A few OUIs, standing in for the server's IEEE registry lookup. */
const VENDORS: Record<string, string> = { "001565": "Yealink", "00156D": "Ubiquiti", "3C2AF4": "Brother", "F4E2C6": "Ubiquiti", "001A11": "Google" };

const lookupVendor: MacVendorLookup = async (oui) => {
    await wait(150);
    return VENDORS[oui];
};

function MacDemo() {
    const form = useAppForm({
        ...formBehaviour,
        defaultValues: { phone: "0015.65aa.bbcc", printer: "3c-2a-f4-11-22-33", laptop: "DA:A1:19:2B:3C:4D" },
        validators: { onDynamic: z.object({ phone: macSchema({ required: true }), printer: macSchema(), laptop: macSchema() }) },
        onSubmit: () => wait(600),
    });
    return (<Form form={form} aria-label="Devices" className="w-full" data-testid="mac-form">
      <FormStack>
        <form.AppField name="phone">{(field) => <field.MacAddressField label="Desk phone" lookupVendor={lookupVendor} width="medium"/>}</form.AppField>
        <form.AppField name="printer">{(field) => <field.MacAddressField label="Printer" lookupVendor={lookupVendor} format="hyphen" width="medium"/>}</form.AppField>
        <form.AppField name="laptop">{(field) => <field.MacAddressField label="Laptop Wi-Fi" lookupVendor={lookupVendor} width="medium"/>}</form.AppField>
        <FormActions>
          <form.SubmitButton>Save</form.SubmitButton>
        </FormActions>
      </FormStack>
    </Form>);
}

export default function MacFieldSection() {
    return (<SectionPage>
      <Specimen title="Any notation in, one out" description="Colons, hyphens (Windows), dots (Cisco) or none; leave the field and it is written one way. With a vendor lookup the hint names the maker; a randomised Wi-Fi address says so.">
        <MacDemo />
      </Specimen>

      <></>

      <></>

      <></>

      <Usage code={`
import { macSchema } from "glass-ux/mac-field"

const schema = z.object({ mac: macSchema({ required: true }) })

<form.AppField name="mac">
  {(field) => <field.MacAddressField label="MAC address" lookupVendor={(oui) => api.vendorOf(oui)} />}
</form.AppField>`}/>

      <></>

      <></>
    </SectionPage>);
}
