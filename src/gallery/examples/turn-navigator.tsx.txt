import { useState } from "react";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { TurnNavigator } from "glass-ux/ai";

export default function TurnNavigatorExample() {
  const [active, setActive] = useState(0);
  return (
    <SectionPage>
      <Specimen title="Navigate conversation turns" layout="stack">
        <TurnNavigator
          turns={[
            { index: 1, preview: "Plan an exhibition" },
            { index: 2, preview: "Review the schedule" },
            { index: 3, preview: "Confirm the rooms" },
          ]}
          active={active}
          onNavigate={setActive}
        />
        <p>Selected turn: {active + 1}</p>
      </Specimen>
    </SectionPage>
  );
}
