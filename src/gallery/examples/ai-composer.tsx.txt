import { useState } from "react"
import { AiHero, AiStatusStrip, Composer, SuggestionChips, type AiAttachment, type EntityRef } from "glass-ux/ai"
import { ContainerLab, Example, Rules, SectionPage, Specimen, TokenTable } from "glass-ux/gallery/kit"
import { toast } from "glass-ux/toast"
import { BUDGET, BUDGET_LOW, BUDGET_OUT, COMMANDS, E, MODEL, NOW, SCOPED_SUGGESTIONS, SUGGESTIONS } from "glass-ux/gallery/sections/ai.fixtures"
import { screenshot } from "glass-ux/gallery/sections/ai.screens"
import { AiGallery, DockedColumn, LabScroll } from "glass-ux/gallery/sections/ai.kit"

const ATTACHMENTS: AiAttachment[] = [
  { id: "a1", name: "printer-error.png", size: 412_000, mime: "image/png", state: "ready", previewUrl: screenshot({ app: "Print queue", hue: 20 }) },
  { id: "a2", name: "NST-C106-system-log.evtx", size: 2_480_000, state: "uploading", progress: 0.42 },
  { id: "a3", name: "spooler-dump.dmp", size: 48_000_000, state: "rejected", reason: "Over the 25 MB limit" },
]

function LiveComposer() {
  const [context, setContext] = useState<EntityRef[]>([E.c106Online])
  const [attachments, setAttachments] = useState<AiAttachment[]>([ATTACHMENTS[0]!])
  const [run, setRun] = useState<"idle" | "sending" | "running" | "stopping">("idle")
  return (
    <Composer
      run={run}
      onSend={async (text) => {
        setRun("sending")
        await new Promise((resolve) => setTimeout(resolve, 500))
        setRun("running")
        toast.info("Sent", { description: text })
      }}
      onStop={() => {
        setRun("stopping")
        setTimeout(() => setRun("idle"), 700)
      }}
      onEditLast={() => toast.info("Editing your last message")}
      context={context}
      onContextChange={setContext}
      model={MODEL}
      onOpenSettings={() => toast.info("AI settings")}
      attachments={attachments}
      onAttach={(files) => setAttachments((current) => [...current, ...files.map((file, index) => ({ id: `${NOW}-${index}-${file.name}`, name: file.name, size: file.size, mime: file.type, state: "ready" as const }))])}
      onRemoveAttachment={(id) => setAttachments((current) => current.filter((item) => item.id !== id))}
      commands={COMMANDS}
    />
  )
}
