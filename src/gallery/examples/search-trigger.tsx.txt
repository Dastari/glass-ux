import { useContext, useState } from "react";
import { DemoScope } from "glass-ux/gallery/demo-scope";
import {
  MOCK_ORGANISATIONS,
  PORTRAITS,
  toContextItem,
} from "glass-ux/gallery/fixtures/organisations";
import { WORKSPACE_SETTINGS } from "glass-ux/gallery/fixtures/brand";
import { Example, SectionPage, Specimen } from "glass-ux/gallery/kit";
import { AccountButton, SearchTrigger, TopBar, TopBarSettingsButton } from "glass-ux/shell";
import { ContextSwitcher } from "glass-ux/shell/context-switcher/index";
import { Breadcrumb } from "glass-ux/breadcrumb";
import { Menu } from "glass-ux/menu";

const SWITCHER_ORGS = MOCK_ORGANISATIONS.map(toContextItem);

function SpecimenTopBar({ label }: { label: string }) {
  const demo = useContext(DemoScope);
  const [scope, setScope] = useState<string | null>("northstar");
  const [unread, setUnread] = useState(3);
  return (
    <TopBar
      scope={<ContextSwitcher items={SWITCHER_ORGS} value={scope} onValueChange={setScope} />}
      breadcrumb={
        <Breadcrumb
          aria-label={`Breadcrumb: ${demo ? `${demo}: ` : ""}${label}`}
          items={[
            { id: "devices", label: "Devices", href: "#devices" },
            { id: "c106", label: "NST-C106", href: "#c106" },
            { id: "overview", label: "Overview" },
          ]}
        />
      }
      search={
        <SearchTrigger
          placeholder="Search devices, organisations, tickets…"
          shortcut="mod+shift+k"
          onOpen={() => {}}
        />
      }
      actions={
        <>
          <TopBarSettingsButton label={WORKSPACE_SETTINGS.label} data-keep="" />
        </>
      }
      account={
        <Menu.Root>
          <Menu.Trigger
            render={
              <AccountButton
                name="Avery Quinn"
                email="avery.quinn@orbit.example.com.au"
                avatarSrc={PORTRAITS[1]}
                unread={unread}
              />
            }
          />
          <Menu.Popup align="end">
            <Menu.Item
              disabledReason={unread === 0 ? "No unread notifications" : undefined}
              onClick={() => setUnread(0)}
            >
              Mark notifications read
            </Menu.Item>
          </Menu.Popup>
        </Menu.Root>
      }
    />
  );
}

/* ── Section ─────────────────────────────────────────────────────────────── */
export default function AppShellSection() {
  return (
    <SectionPage>
      <Specimen
        title="Top bar"
        description="The application context: organisation switcher, search, settings and the account with its notification badge. As the bar narrows, the organisation and account fold to avatars and search to a button."
        wide
        layout="stack"
      >
        {(["wide", "medium", "narrow"] as const).map((width) => (
          <Example
            key={width}
            label={width === "wide" ? "Wide" : width === "medium" ? "Medium" : "Narrow"}
          >
            <div className="gallery-shell-topbar" data-width={width}>
              <SpecimenTopBar label={width} />
            </div>
          </Example>
        ))}
      </Specimen>
    </SectionPage>
  );
}
