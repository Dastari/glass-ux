import { useState } from "react";

import { MOCK_ORGANISATIONS, toSwitcherOrganisation } from "glass-ux/gallery/fixtures/organisations";

import { WORKSPACE_SETTINGS } from "glass-ux/gallery/fixtures/brand";

import { Example, SectionPage, Specimen } from "glass-ux/gallery/kit";

import { AccountButton, SearchTrigger, TopBar, TopBarSettingsButton } from "glass-ux/shell";

import { OrgSwitcher } from "glass-ux/shell/org-switcher/index";

import { Breadcrumb } from "glass-ux/breadcrumb";

const SWITCHER_ORGS = MOCK_ORGANISATIONS.map(toSwitcherOrganisation);

function SpecimenTopBar() {
    const [scope, setScope] = useState<string | null>("northstar");
    return (<TopBar scope={<OrgSwitcher organisations={SWITCHER_ORGS} value={scope} onValueChange={setScope}/>} breadcrumb={<Breadcrumb items={[
                { id: "devices", label: "Devices", href: "#devices" },
                { id: "c106", label: "NST-C106", href: "#c106" },
                { id: "overview", label: "Overview" },
            ]}/>} search={<SearchTrigger placeholder="Search devices, organisations, tickets…" shortcut="mod+shift+k" onOpen={() => { }}/>} actions={<>
          <TopBarSettingsButton label={WORKSPACE_SETTINGS.label} data-keep=""/>
        </>} account={<AccountButton name="Avery Quinn" email="avery.quinn@orbit.example.com.au" unread={3}/>}/>);
}

/* ── Section ─────────────────────────────────────────────────────────────── */
export default function AppShellSection() {
    return (<SectionPage>
      <></>

    <></>

      <></>

      <></>

      <Specimen title="Top bar" description="The application context, on the canvas: nothing in it depends on the organisation. The switcher is the door to the organisation context below; Glass UX settings is the gear; notifications are the badge on the avatar. It adapts to its own width: the organisation folds to its avatar (with +N on its corner), the search field becomes a square and the account folds to its avatar." wide layout="stack">
        {(["wide", "medium", "narrow"] as const).map((width) => (<Example key={width} label={width === "wide" ? "Wide" : width === "medium" ? "Medium" : "Narrow"}>
            <div className="gallery-shell-topbar" data-width={width}>
              <SpecimenTopBar />
            </div>
          </Example>))}
      </Specimen>

      <></>

      <></>

      <></>

      <></>

      <></>
    </SectionPage>);
}
