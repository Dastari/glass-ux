import { useEffect, useMemo, useRef, useState } from "react";
import { IconCaseSensitive, IconRegex, IconWholeWord } from "glass-ux/icons";
import { Button } from "glass-ux/button";
import { FindBar } from "glass-ux/find-bar";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";

const LOG = [
  "09:14:02 harbour-sync started",
  "09:14:03 connected to cedar-archive",
  "09:14:07 copied 214 drawings",
  "09:14:09 warning: cedar-archive is slow",
  "09:14:12 copied 62 photographs",
  "09:14:18 warning: retrying harbour-sync",
  "09:14:21 harbour-sync reconnected",
  "09:14:25 copied 18 notes",
  "09:14:29 verified 294 files",
  "09:14:33 harbour-sync finished",
];

function Toggles({
  matchCase,
  onMatchCase,
  wholeWord,
  onWholeWord,
}: {
  matchCase: boolean;
  onMatchCase: (value: boolean) => void;
  wholeWord: boolean;
  onWholeWord: (value: boolean) => void;
}) {
  return (
    <>
      <Button
        variant={matchCase ? "soft" : "ghost"}
        size="sm"
        icon={<IconCaseSensitive />}
        aria-label="Match case"
        aria-pressed={matchCase}
        onClick={() => onMatchCase(!matchCase)}
      />
      <Button
        variant={wholeWord ? "soft" : "ghost"}
        size="sm"
        icon={<IconWholeWord />}
        aria-label="Whole word"
        aria-pressed={wholeWord}
        onClick={() => onWholeWord(!wholeWord)}
      />
      <Button
        variant="ghost"
        size="sm"
        icon={<IconRegex />}
        aria-label="Pattern"
        disabledReason="Patterns are not available in this log"
      />
    </>
  );
}

/** The host's search: FindBar only hears the results. */
function useLogSearch(term: string, matchCase: boolean, wholeWord: boolean) {
  const lines = useMemo(() => {
    if (!term) return [];
    const wanted = matchCase ? term : term.toLowerCase();
    return LOG.flatMap((line, index) => {
      const text = matchCase ? line : line.toLowerCase();
      const found = wholeWord ? text.split(/\W+/).includes(wanted) : text.includes(wanted);
      return found ? [index] : [];
    });
  }, [term, matchCase, wholeWord]);
  const [cursor, setCursor] = useState(0);
  const current = lines.length ? Math.min(cursor, lines.length - 1) : undefined;
  return {
    lines,
    current,
    next: () => setCursor(((current ?? 0) + 1) % lines.length),
    previous: () => setCursor(((current ?? 0) - 1 + lines.length) % lines.length),
  };
}

function FloatingLog() {
  const [term, setTerm] = useState("harbour");
  const [open, setOpen] = useState(true);
  const [matchCase, setMatchCase] = useState(false);
  const [wholeWord, setWholeWord] = useState(false);
  const opener = useRef<HTMLButtonElement>(null);
  const { lines, current, next, previous } = useLogSearch(term, matchCase, wholeWord);
  const active = current === undefined ? undefined : lines[current];
  const scrollerRef = useRef<HTMLDivElement>(null);
  useEffect(() => {
    // The host reveals its own match, in the scroller alone, leaving room for the bar.
    const scroller = scrollerRef.current;
    const match = scroller?.querySelector<HTMLElement>("[data-match]");
    if (scroller && match)
      scroller.scrollTop =
        match.offsetTop -
        scroller.offsetTop -
        parseFloat(getComputedStyle(scroller).scrollPaddingBlockStart);
  }, [active]);
  return (
    <div className="w-full max-w-2xl" data-testid="find-bar-log">
      <Button ref={opener} size="sm" onClick={() => setOpen(true)}>
        Find in log
      </Button>
      <div className="gallery-find-frame">
        <div
          ref={scrollerRef}
          className="gallery-find-scroller"
          data-testid="find-bar-scroller"
          role="log"
          aria-label="Sync log"
          tabIndex={0}
        >
          {LOG.concat(LOG, LOG).map((line, index) => (
            <div key={index} data-match={index === active ? "" : undefined}>
              {line}
            </div>
          ))}
        </div>
        <FindBar
          floating
          open={open}
          onOpenChange={setOpen}
          returnFocusRef={opener}
          aria-label="Find in log"
          value={term}
          onValueChange={setTerm}
          matchCount={lines.length}
          matchIndex={current}
          onNext={next}
          onPrevious={previous}
          toggles={
            <Toggles
              matchCase={matchCase}
              onMatchCase={setMatchCase}
              wholeWord={wholeWord}
              onWholeWord={setWholeWord}
            />
          }
        />
      </div>
    </div>
  );
}

/** A search that takes three seconds: the count climbs in place. */
function ClimbingCount() {
  const [count, setCount] = useState(0);
  const [searching, setSearching] = useState(false);
  useEffect(() => {
    if (!searching) return;
    const timer = setInterval(() => {
      setCount((value) => {
        if (value >= 8400) {
          setSearching(false);
          return 8412;
        }
        return value + 1400;
      });
    }, 500);
    return () => clearInterval(timer);
  }, [searching]);
  return (
    <div className="flex w-full max-w-xl flex-col items-start gap-3">
      <Button
        size="sm"
        onClick={() => {
          setCount(0);
          setSearching(true);
        }}
      >
        Run the search
      </Button>
      <FindBar
        aria-label="Find in archive"
        defaultValue="cedar"
        matchCount={count}
        matchIndex={count ? 0 : undefined}
        searching={searching}
        onNext={() => {}}
        onPrevious={() => {}}
      />
    </div>
  );
}

function ReplaceDemo() {
  const [text, setText] = useState("Cedar Studio reviewed the Cedar Studio plans.");
  const [term, setTerm] = useState("Cedar");
  const count = term ? text.split(term).length - 1 : 0;
  return (
    <div className="flex w-full max-w-xl flex-col gap-3">
      <FindBar
        aria-label="Find and replace"
        value={term}
        onValueChange={setTerm}
        matchCount={count}
        matchIndex={count ? 0 : undefined}
        onNext={() => {}}
        onPrevious={() => {}}
        defaultReplaceOpen
        defaultReplaceValue="Harbour"
        onReplace={(replacement) => setText(text.replace(term, replacement))}
        onReplaceAll={(replacement) => setText(text.replaceAll(term, replacement))}
      />
      <p data-testid="find-bar-replaced">{text}</p>
    </div>
  );
}
