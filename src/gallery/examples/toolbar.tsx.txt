import { useState } from "react";
import {
  IconAdjust,
  IconBrush,
  IconColumns,
  IconCrop,
  IconCursor,
  IconEdit,
  IconDownload,
  IconFilter,
  IconGrid,
  IconList,
  IconMove,
  IconRefresh,
  IconShare,
  IconSort,
  IconText,
  IconZoomIn,
  IconZoomOut,
} from "glass-ux/icons";
import { Menu } from "glass-ux/menu";
import { Segmented } from "glass-ux/segmented";
import { Toolbar } from "glass-ux/toolbar";
import { ContainerLab, SectionPage, Specimen } from "glass-ux/gallery/kit";

function ReportToolbar() {
  const [view, setView] = useState<"grid" | "list">("grid");
  const [filters, setFilters] = useState(true);
  const [zoom, setZoom] = useState(100);
  return (
    <Toolbar aria-label="Report view">
      <Toolbar.Item priority="always" menu={null}>
        <Segmented
          size="sm"
          iconOnly
          aria-label="Layout"
          value={view}
          onValueChange={setView}
          options={[
            { value: "grid", label: "Grid", icon: <IconGrid /> },
            { value: "list", label: "List", icon: <IconList /> },
          ]}
        />
      </Toolbar.Item>
      <Toolbar.Separator />
      <Toolbar.Button
        label="Filters"
        icon={<IconFilter />}
        pressed={filters}
        onPressedChange={setFilters}
        shortcut="mod+shift+f"
      />
      <Toolbar.Button label="Sort" icon={<IconSort />} onClick={() => {}} />
      <Toolbar.Button label="Columns" icon={<IconColumns />} priority="low" onClick={() => {}} />
      <Toolbar.Separator />
      <Toolbar.Group aria-label="Zoom" priority="low">
        <Toolbar.Button
          label="Zoom out"
          icon={<IconZoomOut />}
          disabled={zoom <= 50}
          onClick={() => setZoom((z) => Math.max(50, z - 25))}
        />
        <Toolbar.Button
          label="Zoom in"
          icon={<IconZoomIn />}
          disabled={zoom >= 200}
          onClick={() => setZoom((z) => Math.min(200, z + 25))}
        />
      </Toolbar.Group>
      <Toolbar.Spacer />
      <Toolbar.Button label="Refresh" icon={<IconRefresh />} priority="low" onClick={() => {}} />
      <Toolbar.Button
        label="Download CSV"
        icon={<IconDownload />}
        priority="low"
        onClick={() => {}}
      />
      <Toolbar.Item
        priority="high"
        menu={
          <Menu.Submenu label="Share">
            <Menu.Item>Copy link</Menu.Item>
            <Menu.Item>Email…</Menu.Item>
          </Menu.Submenu>
        }
      >
        <Menu.Root>
          <Toolbar.Button label="Share" icon={<IconShare />} showLabel render={<Menu.Trigger />} />
          <Menu.Popup align="end">
            <Menu.Item>Copy link</Menu.Item>
            <Menu.Item>Email…</Menu.Item>
          </Menu.Popup>
        </Menu.Root>
      </Toolbar.Item>
    </Toolbar>
  );
}

const BRUSHES = [
  { value: "brush", label: "Brush", icon: <IconBrush />, shortcut: "b" },
  { value: "pencil", label: "Pencil", icon: <IconEdit />, shortcut: "n" },
];

/** A vertical tool rail: one tool chosen at a time, and a flyout of related brushes. */
function ToolRail({ height }: { height?: string }) {
  const [tool, setTool] = useState("move");
  return (
    <div className="flex items-start gap-4">
      <div style={{ height }}>
        <Toolbar aria-label="Tools" orientation="vertical" className="h-full">
          <Toolbar.Group aria-label="Tool" value={tool} onValueChange={setTool}>
            <Toolbar.Button value="move" label="Move" icon={<IconMove />} shortcut="v" />
            <Toolbar.Button value="select" label="Select" icon={<IconCursor />} shortcut="s" />
            <Toolbar.Flyout label="Brush tools" tools={BRUSHES} />
            <Toolbar.Button value="crop" label="Crop" icon={<IconCrop />} shortcut="c" />
            <Toolbar.Button
              value="text"
              label="Text"
              icon={<IconText />}
              shortcut="t"
              priority="low"
            />
          </Toolbar.Group>
          <Toolbar.Separator />
          <Toolbar.Button label="Adjust" icon={<IconAdjust />} priority="low" onClick={() => {}} />
        </Toolbar>
      </div>
      <p className="type-caption text-fg-muted" data-testid="chosen-tool">
        Tool: {tool}
      </p>
    </div>
  );
}

function ToolRow() {
  const [tool, setTool] = useState("select");
  return (
    <Toolbar aria-label="Row tools">
      <Toolbar.Group aria-label="Row tool" value={tool} onValueChange={setTool}>
        <Toolbar.Button value="move" label="Move" icon={<IconMove />} />
        <Toolbar.Button value="select" label="Select" icon={<IconCursor />} />
        <Toolbar.Flyout label="Row brush tools" tools={BRUSHES} />
      </Toolbar.Group>
    </Toolbar>
  );
}

export default function ToolbarSection() {
  return (
    <SectionPage>
      <Specimen
        title="Toolbar"
        description="Arrow keys move between controls; Tab leaves the row. Toggles that are on take the accent's soft fill."
        layout="stack"
        wide
      >
        <ReportToolbar />
      </Specimen>
      <Specimen
        title="Vertical tool rail"
        description="Up and Down Arrow move between tools, Space or Enter chooses one, and exactly one is chosen (a radio group). The Brush button holds related tools: hold it, press Right Arrow (Left in right-to-left layouts) or use the context-menu key to open them beside the rail."
        layout="row"
      >
        <ToolRail />
      </Specimen>
      <Specimen
        title="Folding by height"
        description="A vertical toolbar folds low-priority tools into More by the height it is given."
        layout="row"
      >
        <ToolRail height="calc(var(--control-h-sm) * 4)" />
      </Specimen>
      <Specimen
        title="Single choice in a row"
        description="The same group in a horizontal toolbar; the flyout opens below its button on Arrow Down."
        layout="row"
      >
        <ToolRow />
      </Specimen>
      <ContainerLab
        title="Folding"
        description="Low-priority items fold into More first, the last ones first; groups fold together; separators go with them. The layout switch never folds."
      >
        {() => (
          <div className="flex h-full items-start p-2">
            <div className="w-full">
              <ReportToolbar />
            </div>
          </div>
        )}
      </ContainerLab>
    </SectionPage>
  );
}
