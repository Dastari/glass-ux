import { Example, SectionPage, Specimen } from "glass-ux/gallery/kit";

import { IconCopy, IconFullscreen, IconKeyboard, IconRefresh, IconRemoteControl, IconSettings, IconWake } from "glass-ux/icons";

import { Button } from "glass-ux/button";

import { Tooltip } from "glass-ux/tooltip";

const TOOLBAR = [
    { label: "Remote control", icon: <IconRemoteControl />, shortcut: "mod+shift+c" },
    { label: "Send Ctrl+Alt+Del", icon: <IconKeyboard /> },
    { label: "Copy clipboard to endpoint", icon: <IconCopy /> },
    { label: "Full screen", icon: <IconFullscreen />, shortcut: "f11" },
    { label: "Session settings", icon: <IconSettings /> },
];

export default function TooltipSection() {
    return (<SectionPage>
      <Specimen title="Live" description="Hover or focus a button. The first tooltip waits 500ms; moving along the toolbar shows the next one at once. Tooltips are an inverted island: dark on a light page, light on a dark one.">
        <Example label="Toolbar (grouped, instant after the first)">
          <div className="glass-control flex gap-1 rounded-control p-1" role="toolbar" aria-label="Remote session">
            {TOOLBAR.map((item) => (<Button key={item.label} variant="ghost" size="sm" icon={item.icon} aria-label={item.label} shortcut={item.shortcut}/>))}
          </div>
        </Example>
        <Example label="Sides">
          {(["top", "right", "bottom", "left"] as const).map((side) => (<Tooltip key={side} content="Last check-in 2 min ago" side={side}>
              <Button size="sm" variant="ghost">
                {side}
              </Button>
            </Tooltip>))}
        </Example>
        <Example label="Disabled — the tooltip gives the reason">
          <Button size="sm" icon={<IconWake />} disabledReason="Endpoint is online">
            Wake on LAN
          </Button>
          <Button size="sm" variant="ghost" icon={<IconRefresh />} aria-label="Refresh inventory" disabledReason="Inventory refreshes every 15 min"/>
        </Example>
      </Specimen>












    </SectionPage>);
}
