import type { ComponentProps } from "react";

import { RadialBar, type ChartState } from "glass-ux/charts";

import { ContainerLab, SectionPage, Specimen, TierBox, TierRow } from "glass-ux/gallery/kit";

import { SkeletonSwaps } from "glass-ux/gallery/sections/charts.kit";

const sample = {
    data: [
        { key: "design", label: "Design", value: 72, max: 100 },
        { key: "build", label: "Build", value: 54, max: 100 },
        { key: "review", label: "Review", value: 36, max: 100 },
    ],
    unit: "%",
} satisfies Omit<ComponentProps<typeof RadialBar>, "label">;

const render = (state: ChartState = "ready") => (<RadialBar {...sample} label="RadialBar · fictional studio projects" title="Studio projects" state={state} emptyText="No project measurements yet" errorText="Measurements could not load" onRetry={() => undefined}/>);

export default function RadialBarSection() {
    return (<SectionPage>
      <Specimen title="RadialBar" description="A focused example using fictional studio project measurements. Hover or use the keyboard to inspect values; the chart's table preserves the exact data." wide>
        <div className="charts-pane" data-testid="chart-radial-bar">
          {render()}
        </div>
      </Specimen>
      <Specimen title="Responsive sizes" description="The same RadialBar adapts to row, tile and full-panel sizes." wide>
        <TierRow>{() => render()}</TierRow>
      </Specimen>
      <Specimen title="Loading, empty and error" description="States stay in the chart's own box. Swap the skeleton to compare its dimensions." wide>
        <SkeletonSwaps id="radial-bar" render={render}/>
        <div className="gallery-tiers">
          <TierBox size="compact" label="Empty">
            {render("empty")}
          </TierBox>
          <TierBox size="full" label="Error">
            {render("error")}
          </TierBox>
        </div>
      </Specimen>
      <ContainerLab>{() => render()}</ContainerLab>
    </SectionPage>);
}
