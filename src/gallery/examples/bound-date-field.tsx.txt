import { z } from "zod";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { addDays, dateSchema, todayIso, type IsoDate } from "glass-ux/date-picker";
import { Form, FormActions, FormStack, formBehaviour, useAppForm } from "glass-ux/form";
import { wait } from "glass-ux/gallery/sections/forms.fixtures";

const today = todayIso();

function DatesDemo() {
  const form = useAppForm({
    ...formBehaviour,
    defaultValues: { review: addDays(today, 7) as IsoDate | null, expires: null as IsoDate | null },
    validators: {
      onDynamic: z.object({
        review: dateSchema({ required: true, min: today }),
        expires: dateSchema(),
      }),
    },
    onSubmit: () => wait(600),
  });
  return (
    <Form form={form} aria-label="Contract dates" className="w-full" data-testid="date-form">
      <FormStack>
        <form.AppField name="review">
          {(field) => <field.DateField label="Next review" required min={today} />}
        </form.AppField>
        <form.AppField name="expires">
          {(field) => <field.DateField label="Contract expires" optional />}
        </form.AppField>
        <FormActions>
          <form.SubmitButton>Save</form.SubmitButton>
        </FormActions>
      </FormStack>
    </Form>
  );
}

export default function DatePickerSection() {
  return (
    <SectionPage>
      <Specimen
        title="In a form"
        description="field.DateField holds a calendar day (2026-10-21). Text that is not a date reaches the form as typed, and a day outside the bounds as read, so dateSchema says which and Save focuses it."
        layout="stack"
      >
        <DatesDemo />
      </Specimen>
    </SectionPage>
  );
}
