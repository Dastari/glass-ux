import { useState, type ReactNode } from "react";
import { IconRefresh } from "glass-ux/icons";
import { Button } from "glass-ux/button";
import { Image } from "glass-ux/image";
import { PORTRAITS } from "glass-ux/gallery/fixtures/organisations";
import { BROKEN_IMAGE, SCENE_COLOURS, SCENE_IMAGES } from "glass-ux/gallery/fixtures/scenes";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";

const RATIOS = [
  ["square", SCENE_IMAGES.ridge],
  ["landscape", SCENE_IMAGES.harbour],
  ["wide", SCENE_IMAGES.dunes],
  ["portrait", SCENE_IMAGES.dusk],
] as const;

function Labelled({ label, children }: { label: string; children: ReactNode }) {
  return (
    <figure className="flex min-w-0 flex-col gap-2">
      {children}
      <figcaption className="type-caption text-fg-muted">{label}</figcaption>
    </figure>
  );
}

export default function ImageSection() {
  const [replay, setReplay] = useState(0);
  return (
    <SectionPage>
      <Specimen
        title="Ratios"
        description="The frame is reserved before the picture arrives, so nothing below it moves."
        layout="stack"
        wide
      >
        <div className="grid grid-cols-2 items-start gap-4 md:grid-cols-4">
          {RATIOS.map(([ratio, src]) => (
            <Labelled key={ratio} label={ratio}>
              <Image src={src} alt={`Sample landscape, ${ratio}`} ratio={ratio} />
            </Labelled>
          ))}
        </div>
      </Specimen>
      <Specimen
        title="States"
        description="A shimmer, or the picture's colour, fills the frame while it loads, and the picture fades in. A missing or failed picture shows a fallback in the icon language."
        layout="stack"
        wide
      >
        <div className="grid grid-cols-2 items-start gap-4 md:grid-cols-4">
          <Labelled label="Loaded">
            <Image key={replay} src={SCENE_IMAGES.harbour} alt="Harbour at midday" />
          </Labelled>
          <Labelled label="Colour, contained">
            <Image
              key={replay}
              src={SCENE_IMAGES.dunes}
              alt="Dunes"
              color={SCENE_COLOURS.dunes}
              fit="contain"
              ratio="wide"
            />
          </Labelled>
          <Labelled label="No source">
            <Image alt="Site photo" />
          </Labelled>
          <Labelled label="Failed">
            <Image src={BROKEN_IMAGE} alt="Pier camera, 06:00" />
          </Labelled>
        </div>
        <div>
          <Button
            size="sm"
            variant="ghost"
            icon={<IconRefresh />}
            onClick={() => setReplay((count) => count + 1)}
          >
            Replay the fade
          </Button>
        </div>
      </Specimen>
      <Specimen
        title="Fallback sizes"
        description="The glyph scales with the frame; the alt text joins it once the frame is 12rem wide."
        layout="stack"
        wide
      >
        <div className="flex flex-wrap items-end gap-4">
          <div className="w-16">
            <Image alt="Thumbnail" ratio="square" />
          </div>
          <div className="w-32">
            <Image src={BROKEN_IMAGE} alt="Gate sensor" ratio="square" />
          </div>
          <div className="w-72 max-w-full">
            <Image src={BROKEN_IMAGE} alt="Loading bay, east camera" />
          </div>
        </div>
      </Specimen>
      <Specimen
        title="Hover inside a link"
        description="Only an image inside a link or button reacts, when that is hovered or focused. Still and reduced motion turn the effects off."
        layout="stack"
        wide
      >
        <div className="grid grid-cols-1 items-start gap-4 sm:grid-cols-3">
          {(["none", "zoom", "sheen"] as const).map((hover, index) => (
            <a
              key={hover}
              href={`#hover-${hover}`}
              className="flex flex-col gap-2 rounded-[var(--radius-media)]"
              onClick={(event) => event.preventDefault()}
            >
              <Image
                src={[SCENE_IMAGES.ridge, SCENE_IMAGES.harbour, SCENE_IMAGES.dusk][index]}
                alt=""
                hover={hover}
              />
              <span className="type-caption text-fg-muted">{hover}</span>
            </a>
          ))}
        </div>
      </Specimen>
      <Specimen
        title="Fit and focus"
        description="cover fills and crops around position; contain shows the whole picture on its colour."
        layout="stack"
        wide
      >
        <div className="grid grid-cols-2 items-start gap-4 md:grid-cols-4">
          <Labelled label="cover">
            <Image src={PORTRAITS[0]} alt="Sample portrait" ratio="wide" />
          </Labelled>
          <Labelled label="cover, position 50% 20%">
            <Image src={PORTRAITS[0]} alt="Sample portrait" ratio="wide" position="50% 20%" />
          </Labelled>
          <Labelled label="contain">
            <Image src={PORTRAITS[0]} alt="Sample portrait" ratio="wide" fit="contain" />
          </Labelled>
          <Labelled label="no rim">
            <Image src={SCENE_IMAGES.harbour} alt="Harbour" frame={false} />
          </Labelled>
        </div>
      </Specimen>
    </SectionPage>
  );
}
