import { Badge } from "glass-ux/badge";
import { Grid, Stack } from "glass-ux/layout";
import { Surface } from "glass-ux/surface";
import { Text } from "glass-ux/text";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";

const SITES = [
  "Parramatta depot",
  "Hobart office",
  "Perth warehouse",
  "Geelong lab",
  "Cairns kiosk",
  "Darwin yard",
];

export default function LayoutSection() {
  return (
    <SectionPage>
      <Specimen
        title="Grid"
        description="As many columns as fit at a minimum width, so tiles reflow with their container, not the window."
        layout="stack"
        wide
      >
        <Grid columns={{ min: "12rem" }}>
          {" "}
          {/* tokens:allow a tile width */}
          {SITES.map((site, index) => (
            <Surface key={site} material="panel" padding="tile" radius="tile">
              <Stack gap="inline">
                <Text variant="label">{site}</Text>
                <Stack direction="row" gap="inline">
                  <Badge size="sm" tone={index % 3 ? "success" : "warning"}>
                    {index % 3 ? "Online" : "Degraded"}
                  </Badge>
                  <Text as="span" variant="caption" tone="subtle" numeric>
                    {12 + index * 7} devices
                  </Text>
                </Stack>
              </Stack>
            </Surface>
          ))}
        </Grid>
      </Specimen>
    </SectionPage>
  );
}
