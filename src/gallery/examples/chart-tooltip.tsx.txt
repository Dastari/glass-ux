import { useRef } from "react";

import { SectionPage, Specimen } from "glass-ux/gallery/kit";

import { ChartTooltip, TooltipBody, useHoverStore } from "glass-ux/charts";

import { Button } from "glass-ux/button";

export default function ChartTooltipExample() {
    const anchor = useRef<HTMLDivElement>(null);
    const store = useHoverStore();
    return (<SectionPage>
      <Specimen title="Plot readout" description="Hover the stage or focus the button to open the chart tooltip." layout="stack">
        <div ref={anchor} className="relative h-45 w-full" onPointerMove={(event) => store.set({
            index: 0,
            x: event.clientX,
            y: event.clientY,
            via: "pointer",
        })} onPointerLeave={() => store.set(null)}>
          <Button onFocus={(event) => {
            const box = event.currentTarget.getBoundingClientRect();
            store.set({
                index: 0,
                x: box.left,
                y: box.bottom,
                via: "keyboard",
            });
        }} onBlur={() => store.set(null)}>
            Week 1 · 42 reservations
          </Button>
        </div>
        <ChartTooltip anchor={anchor} store={store}>
          {() => (<TooltipBody title="Week 1" rows={[
                { key: "booked", label: "Reservations", value: "42", slot: 1 },
            ]}/>)}
        </ChartTooltip>
      </Specimen>
    </SectionPage>);
}
