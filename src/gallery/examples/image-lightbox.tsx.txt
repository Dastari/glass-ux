import { useState } from "react";

import { SectionPage, Specimen } from "glass-ux/gallery/kit";

import { AiGallery } from "glass-ux/gallery/sections/ai.kit";

import { RESULTS } from "glass-ux/gallery/sections/ai.fixtures";

import { ImageLightbox } from "glass-ux/ai";

import { Button } from "glass-ux/button";

export default function ImageLightboxExample() {
    const [open, setOpen] = useState(false);
    const [index, setIndex] = useState(0);
    return (<AiGallery>
      <SectionPage>
        <Specimen title="Inspect sample images" layout="stack">
          <Button onClick={() => setOpen(true)}>Open image viewer</Button>
          <ImageLightbox images={RESULTS.screens.images} index={index} onIndexChange={setIndex} open={open} onOpenChange={setOpen}/>
        </Specimen>
      </SectionPage>
    </AiGallery>);
}
