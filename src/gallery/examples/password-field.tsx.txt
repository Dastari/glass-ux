import { z } from "zod";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { Form, FormActions, FormStack, formBehaviour, useAppForm } from "glass-ux/form";
import { currentPasswordSchema, passwordSchema } from "glass-ux/password-field";
import { wait } from "glass-ux/gallery/sections/forms.fixtures";

const USER = ["Avery Quinn", "avery.quinn@orbit.example.com.au", "Orbit Workspace"];
const schema = z.object({
  current: currentPasswordSchema(),
  next: passwordSchema({ userInputs: USER }),
});

function ChangePasswordDemo() {
  const form = useAppForm({
    ...formBehaviour,
    defaultValues: { current: "", next: "" },
    validators: { onDynamic: schema },
    onSubmit: () => wait(700),
  });
  return (
    <Form form={form} aria-label="Change password" className="w-full" data-testid="password-form">
      <FormStack>
        <form.AppField name="current">
          {(field) => (
            <field.PasswordField label="Current password" purpose="current" width="medium" />
          )}
        </form.AppField>
        <form.AppField name="next">
          {(field) => (
            <field.PasswordField label="New password" generate userInputs={USER} width="medium" />
          )}
        </form.AppField>
        <FormActions>
          <form.CancelButton />
          <form.SubmitButton>Change password</form.SubmitButton>
        </FormActions>
      </FormStack>
    </Form>
  );
}

export default function PasswordFieldSection() {
  return (
    <SectionPage>
      <Specimen
        title="In a form"
        description="currentPasswordSchema only requires a value; passwordSchema wants 12 characters and at least Fair, and penalises the person's own name and email."
      >
        <ChangePasswordDemo />
      </Specimen>
    </SectionPage>
  );
}
