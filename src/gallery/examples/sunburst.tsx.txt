import type { ComponentProps } from "react";

import { Sunburst, type ChartState } from "glass-ux/charts";

import { ContainerLab, SectionPage, Specimen, TierBox, TierRow } from "glass-ux/gallery/kit";

import { SkeletonSwaps } from "glass-ux/gallery/sections/charts.kit";

const sample = {
    data: {
        label: "Studio",
        children: [
            {
                label: "Design",
                children: [
                    { label: "Planning", value: 32 },
                    { label: "Review", value: 20 },
                ],
            },
            { label: "Delivery", value: 48 },
        ],
    },
} satisfies Omit<ComponentProps<typeof Sunburst>, "label">;

const render = (state: ChartState = "ready") => (<Sunburst {...sample} label="Sunburst · fictional studio projects" title="Studio projects" state={state} emptyText="No project measurements yet" errorText="Measurements could not load" onRetry={() => undefined}/>);

export default function SunburstSection() {
    return (<SectionPage>
      <Specimen title="Sunburst" description="A focused example using fictional studio project measurements. Hover or use the keyboard to inspect values; the chart's table preserves the exact data." wide>
        <div className="charts-pane" data-testid="chart-sunburst">
          {render()}
        </div>
      </Specimen>
      <Specimen title="Responsive sizes" description="The same Sunburst adapts to row, tile and full-panel sizes." wide>
        <TierRow>{() => render()}</TierRow>
      </Specimen>
      <Specimen title="Loading, empty and error" description="States stay in the chart's own box. Swap the skeleton to compare its dimensions." wide>
        <SkeletonSwaps id="sunburst" render={render}/>
        <div className="gallery-tiers">
          <TierBox size="compact" label="Empty">
            {render("empty")}
          </TierBox>
          <TierBox size="full" label="Error">
            {render("error")}
          </TierBox>
        </div>
      </Specimen>
      <ContainerLab>{() => render()}</ContainerLab>
    </SectionPage>);
}
