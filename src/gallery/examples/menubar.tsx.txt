const after = <T,>(ms: number, signal: AbortSignal, value: () => T) =>
  new Promise<T>((resolve, reject) => {
    const timer = setTimeout(() => {
      try {
        resolve(value());
      } catch (error) {
        reject(error instanceof Error ? error : new Error("Load failed"));
      }
    }, ms);
    signal.addEventListener("abort", () => clearTimeout(timer));
  });

function LoadedBar() {
  const attempts = useRef(0);
  const menus: MenubarMenu[] = [
    {
      id: "documents",
      label: "Documents",
      items: [
        {
          kind: "submenu",
          id: "recent",
          label: "Open recent",
          loadingRows: 3,
          loadItems: ({ signal }) => after(1500, signal, () => RECENT),
        },
        {
          kind: "submenu",
          id: "shared",
          label: "Shared with me",
          loadingRows: 2,
          loadError: "Shared documents are unavailable. Try again.",
          loadItems: ({ signal }) =>
            after(800, signal, () => {
              attempts.current += 1;
              if (attempts.current % 2 === 1) throw new Error("503");
              return [{ id: "minutes", label: "Meeting minutes" }];
            }),
        },
        {
          kind: "submenu",
          id: "archived",
          label: "Archived",
          emptyLabel: "No archived documents",
          loadItems: ({ signal }) => after(300, signal, () => []),
        },
      ],
    },
    {
      id: "templates",
      label: "Templates",
      loadingRows: 3,
      loadItems: ({ signal }) =>
        after(1500, signal, () =>
          ["Report", "Memo", "Letter"].map((label) => ({ id: label, label })),
        ),
    },
  ];
  return (
    <div className="w-full" data-testid="menubar-loaded">
      <Menubar aria-label="Documents" menus={menus} />
    </div>
  );
}

function ControlledBar() {
  const [open, setOpen] = useState<string | null>(null);
  return (
    <div className="flex w-full flex-col items-start gap-2" data-testid="menubar-controlled">
      <Menubar
        aria-label="Project"
        menus={BOX_MENUS.slice(0, 3)}
        openMenu={open}
        onOpenMenuChange={setOpen}
      />
      <div className="flex items-center gap-2">
        <Button size="sm" onClick={() => setOpen("view")}>
          Open View
        </Button>
        <span className="type-caption" data-testid="menubar-open-value">
          Open: {open ?? "none"}

      </div>

}

function ActionsBar() {
  const actions = useMenubarActions(reportActions, REPORTS);
  const menus: MenubarMenu[] = [
    {
      id: "report",
      label: "Report",
      items: [
        actions.item("save"),
        actions.item("publish"),
        { kind: "separator", id: "danger-separator" },
        actions.item("delete"),
      ],
    },
    { id: "all", label: "All actions", items: actions.all },
  ];
  return <Menubar aria-label="Report actions" menus={menus} />;
}

interface Report {
  name: string;
  published: boolean;
}

const REPORTS: readonly Report[] = [{ name: "Harbour survey", published: true }];
const reportActions = defineActions<Report>("report", [
  {
    id: "save",
    label: "Save",
    icon: IconSave,
    group: "manage",
    shortcut: "mod+s",
    run: () => undefined,
  },
  {
    id: "publish",
    label: "Publish",
    group: "manage",
    shortcut: "mod+k mod+p",
    disabledReason: ([report]) => report?.published && "Already published",
    run: () => undefined,
  },
  {
    id: "delete",
    label: "Delete",
    icon: IconTrash,
    group: "danger",
    confirm: ([report]) => ({
      title: `Delete ${report?.name}?`,
      body: "This cannot be undone.",
      confirmLabel: "Delete",
      tone: "danger",
    }),
    run: () => undefined,
  },
]);

function RebindingBar() {
  const keymap = useMemo(() => ({ "report.save": ["mod+shift+s"] }), []);
  return (
    <KeymapProvider keymap={keymap}>
      <ActionsBar />
    </KeymapProvider>
  );
}
