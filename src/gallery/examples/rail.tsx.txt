import { useContext, useState, type ReactNode } from "react";
import { DemoScope } from "glass-ux/gallery/demo-scope";
import { ORGANISATION_SETTINGS, moduleRail } from "glass-ux/gallery/fixtures/brand";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import {
  IconArchive,
  IconCalendar,
  IconFolder,
  IconHistory,
  IconInbox,
  IconNewChat,
  IconOrganisation,
  IconReporting,
  IconSearch,
  IconSettings,
  IconUsers,
} from "glass-ux/icons";
import { AppShell, Rail, SectionDrawer, ShellBrand } from "glass-ux/shell";
import { CountBadge } from "glass-ux/badge";
import { BrandMark } from "glass-ux/brand";
import { Button } from "glass-ux/button";
import { EmptyState } from "glass-ux/empty-state";
import { Segmented } from "glass-ux/segmented";

/* ── Live parts ──────────────────────────────────────────────────────────── */
const RAIL_ITEMS: Array<{
  label: string;
  icon: ReactNode;
  shortcut: string;
}> = moduleRail("rmm").map((item) => ({
  label: item.label,
  icon: <item.icon />,
  shortcut: item.shortcut,
}));
const SPECIMEN_ITEMS = [
  { label: "Conversation history", icon: <IconHistory />, count: 0 },
  { label: "Archived chats", icon: <IconArchive />, count: 6 },
];
const SPECIMEN_PROJECTS = ["Northstar Studio migration", "Patch Tuesday · October"];
const OVERFLOW_ITEMS = [
  { label: "Projects", icon: <IconFolder /> },
  { label: "Search", icon: <IconSearch /> },
  { label: "History", icon: <IconHistory /> },
  { label: "Archive", icon: <IconArchive /> },
  { label: "Calendar", icon: <IconCalendar /> },
  { label: "Reports", icon: <IconReporting /> },
  { label: "Team", icon: <IconUsers /> },
  { label: "Inbox", icon: <IconInbox /> },
];

function RailOverflowFrame({ labels = false }: { labels?: boolean }) {
  const [current, setCurrent] = useState("Projects");
  const [overflow, setOverflow] = useState<"more" | "scroll">("more");
  const scope = useContext(DemoScope);
  const name = labels ? "Rail labels (specimen)" : "Rail overflow (specimen)";
  return (
    <div
      className="gallery-shell-rail-frame"
      data-testid={labels ? "rail-labels" : "rail-overflow"}
    >
      <Rail.Root
        aria-label={scope ? `${name}: ${scope}` : name}
        overflow={overflow}
        labels={labels}
      >
        {OVERFLOW_ITEMS.map((item) => (
          <Rail.Item
            key={item.label}
            icon={item.icon}
            label={item.label}
            active={item.label === current}
            badge={
              item.label === "Inbox" ? (
                <CountBadge value={4} tone="accent" overlay aria-hidden />
              ) : undefined
            }
            onClick={() => setCurrent(item.label)}
          />
        ))}
        <Rail.Spacer />
        <Rail.Item
          icon={<IconSettings />}
          label="Settings"
          onClick={() => setCurrent("Settings")}
          active={current === "Settings"}
        />
      </Rail.Root>
      <div className="gallery-shell-rail-frame-side">
        <Segmented
          size="sm"
          aria-label="Overflow"
          value={overflow}
          onValueChange={setOverflow}
          options={[
            { value: "more", label: "More" },
            { value: "scroll", label: "Scroll" },
          ]}
        />
        <p className="type-caption text-fg-muted">{current}</p>
      </div>
    </div>
  );
}

/** Rail labels in a working shell: the rail's column, the mark above it and the drawer beside it. */
function LabelledRailShell() {
  const scope = useContext(DemoScope);
  const prefix = scope ? `${scope}: ` : "";
  return (
    <div className="gallery-shell-labelled" data-testid="rail-labels-shell">
      <AppShell
        contentAs="div"
        className="h-full!"
        brand={<ShellBrand label="Meridian" href="#examples" mark={<IconOrganisation />} />}
        rail={
          <Rail.Root aria-label={`${prefix}Labelled rail sample navigation`} labels>
            <Rail.Item icon={<IconFolder />} label="Projects" active />
            <Rail.Item icon={<IconCalendar />} label="Calendar" />
            <Rail.Item
              icon={<IconInbox />}
              label="Inbox"
              badge={<CountBadge value={4} tone="accent" overlay aria-hidden />}
            />
            <Rail.Item icon={<IconOrganisation />} label="Organisations" />
            <Rail.Spacer />
            <Rail.Item icon={<IconSettings />} label="Settings" />
          </Rail.Root>
        }
        drawer={
          <SectionDrawer.Root aria-label={`${prefix}Labelled rail sample sections`}>
            <SectionDrawer.Header title="Projects" />
            <SectionDrawer.Body>
              <SectionDrawer.List>
                <SectionDrawer.Item label="Exhibition" count={3} active />
                <SectionDrawer.Item label="Archive" />
              </SectionDrawer.List>
            </SectionDrawer.Body>
          </SectionDrawer.Root>
        }
        topBar={<span>Projects / Exhibition</span>}

      </AppShell>
    </div>
  );
}

function RailAndDrawer() {
  const scope = useContext(DemoScope);
  // SectionDrawer.Search: local state, the host filters its own rows and highlights the match.
  const [query, setQuery] = useState("");
  const needle = query.trim().toLocaleLowerCase("en-AU");
  const keep = (label: string) => !needle || label.toLocaleLowerCase("en-AU").includes(needle);
  const items = SPECIMEN_ITEMS.filter((item) => keep(item.label));
  const projects = SPECIMEN_PROJECTS.filter(keep);
  return (
    <div className="gallery-shell-sidebar">
      <Rail.Root aria-label={scope ? `Main (specimen): ${scope}` : "Main (specimen)"}>
        {RAIL_ITEMS.map((item) => (
          <Rail.Item
            key={item.label}
            icon={item.icon}
            label={item.label}
            shortcut={item.shortcut}
            active={item.label === "Assistant"}
          />
        ))}
        <Rail.Spacer />
        <Rail.Item
          icon={<IconSettings />}
          label={ORGANISATION_SETTINGS.label}
          shortcut={ORGANISATION_SETTINGS.shortcut}
        />
      </Rail.Root>
      <SectionDrawer.Root
        aria-label={
          scope ? `Assistant navigation (specimen): ${scope}` : "Assistant navigation (specimen)"
        }
      >
        <SectionDrawer.Header icon={<BrandMark mark="ai" />} title="Assistant" />
        <SectionDrawer.Search noun="Assistant" value={query} onValueChange={setQuery} />
        <SectionDrawer.Body>
          <Button variant="primary" block icon={<IconNewChat />}>
            New chat
          </Button>
          {items.length ? (
            <SectionDrawer.List>
              {items.map((item) => (
                <SectionDrawer.Item
                  key={item.label}
                  icon={item.icon}
                  label={<SectionDrawer.Match text={item.label} query={query} />}
                  count={item.count}
                />
              ))}
            </SectionDrawer.List>
          ) : null}
          {projects.length ? (
            <SectionDrawer.Group id="gallery.ai.projects" title="Projects">
              <SectionDrawer.List>
                {projects.map((label) => (
                  <SectionDrawer.Item
                    key={label}
                    icon={<IconFolder />}
                    label={<SectionDrawer.Match text={label} query={query} />}
                    active={label === SPECIMEN_PROJECTS[0]}
                  />
                ))}
              </SectionDrawer.List>
            </SectionDrawer.Group>
          ) : null}
          {!items.length && !projects.length ? (
            <EmptyState icon={<IconSearch />}>{`No items match "${query.trim()}"`}</EmptyState>
          ) : null}
        </SectionDrawer.Body>
        <SectionDrawer.Footer>
          <SectionDrawer.List>
            <SectionDrawer.Item icon={<IconSettings />} label="AI settings" />
          </SectionDrawer.List>
        </SectionDrawer.Footer>
      </SectionDrawer.Root>
    </div>
  );
}

/* ── Section ─────────────────────────────────────────────────────────────── */

  return (
    <SectionPage>
      <Specimen

        <RailOverflowFrame />
      </Specimen>

      <Specimen
        title="Rail labels"
        description="labels on Rail.Root draws every item's name under its icon, More included. Items that do not fit still move into More, and a name too long for the column ends in an ellipsis with the full name in its tooltip."

        <RailOverflowFrame labels />
      </Specimen>

      <Specimen
        title="Rail labels in the shell"
        description="The shell's rail column follows the labelled rail's width, with the mark above it."
        wide
        layout="stack"
      >
        <LabelledRailShell />
      </Specimen>
    </SectionPage>
  );
}
