import { useState } from "react";
import { Example, SectionPage, Specimen } from "glass-ux/gallery/kit";
import {
  IconChevronDown,
  IconColumns,
  IconCopy,
  IconDownload,
  IconEdit,
  IconExternalLink,
  IconMoreHorizontal,
  IconPin,
  IconRefresh,
  IconTrash,
} from "glass-ux/icons";
import { ActionMenu, ActionProvider } from "glass-ux/actions";
import { Button } from "glass-ux/button";
import { Menu } from "glass-ux/menu";
import { ENDPOINTS, endpointActions } from "glass-ux/gallery/sections/overlays.fixtures";

const nst = ENDPOINTS[0]!;
const win11 = ENDPOINTS[2]!;

function ViewMenu() {
  const [columns, setColumns] = useState({ user: true, organisation: true, lastSeen: false });
  const [density, setDensity] = useState("comfortable");
  return (
    <Menu.Root>
      <Menu.Trigger
        render={
          <Button icon={<IconColumns />} iconEnd={<IconChevronDown />}>
            View
          </Button>
        }
      />
      <Menu.Popup>
        <Menu.Group>
          <Menu.GroupLabel>Columns</Menu.GroupLabel>
          <Menu.CheckboxItem
            checked={columns.user}
            onCheckedChange={(user) => setColumns((c) => ({ ...c, user }))}
          >
            Signed-in user
          </Menu.CheckboxItem>
          <Menu.CheckboxItem
            checked={columns.organisation}
            onCheckedChange={(organisation) => setColumns((c) => ({ ...c, organisation }))}
          >
            Organisation
          </Menu.CheckboxItem>
          <Menu.CheckboxItem
            checked={columns.lastSeen}
            onCheckedChange={(lastSeen) => setColumns((c) => ({ ...c, lastSeen }))}
          >
            Last seen
          </Menu.CheckboxItem>
        </Menu.Group>
        <Menu.Separator />
        <Menu.RadioGroup value={density} onValueChange={(value) => setDensity(String(value))}>
          <Menu.GroupLabel>Density</Menu.GroupLabel>
          <Menu.RadioItem value="comfortable">Comfortable</Menu.RadioItem>
          <Menu.RadioItem value="compact">Compact</Menu.RadioItem>
        </Menu.RadioGroup>
      </Menu.Popup>
    </Menu.Root>
  );
}

function FileMenu() {
  return (
    <Menu.Root>
      <Menu.Trigger
        render={
          <Button
            variant="ghost"
            icon={<IconMoreHorizontal />}
            aria-label="Actions for Q3 patch report.pdf"
          />
        }
      />
      <Menu.Popup align="end">
        <Menu.Group>
          <Menu.Item icon={<IconExternalLink />} shortcut="mod+enter">
            Open in a new window
          </Menu.Item>
          <Menu.Item icon={<IconDownload />} shortcut="mod+s">
            Download
          </Menu.Item>
          <Menu.Item icon={<IconCopy />} shortcut="mod+c">
            Copy link
          </Menu.Item>
        </Menu.Group>
        <Menu.Group>
          <Menu.Item icon={<IconEdit />} shortcut="f2">
            Rename
          </Menu.Item>
          <Menu.Item icon={<IconPin />}>Pin to dashboard</Menu.Item>
          <Menu.Item icon={<IconRefresh />} disabledReason="Report is still generating">
            Regenerate
          </Menu.Item>
        </Menu.Group>
        <Menu.Separator />
        <Menu.Group>
          <Menu.Item icon={<IconTrash />} tone="danger" shortcut="delete">
            Delete report
          </Menu.Item>
        </Menu.Group>
      </Menu.Popup>
    </Menu.Root>
  );
}

export default function MenuSection() {
  return (
    <ActionProvider>
      <SectionPage>
        <Specimen
          title="Live"
          description="Arrow keys move, → and ← open and close submenus, typing jumps to a row, Enter runs it, Esc closes and focus returns to the trigger."
        >
          <Example label="Endpoint actions (online)">
            <ActionMenu set={endpointActions} targets={[nst]} label={nst.name} size="md" />
          </Example>
          <Example label="Offline endpoint">
            <ActionMenu set={endpointActions} targets={[win11]} label={win11.name} size="md" />
          </Example>
          <Example label="Shortcuts, disabled reason">
            <FileMenu />
          </Example>
          <Example label="Toggles and choices">
            <ViewMenu />
          </Example>
        </Specimen>
      </SectionPage>
    </ActionProvider>
  );
}
