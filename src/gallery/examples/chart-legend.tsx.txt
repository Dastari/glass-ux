import { useState } from "react";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { ChartLegend } from "glass-ux/charts";

export default function ChartLegendExample() {
  const [hidden, setHidden] = useState<ReadonlySet<string>>(new Set());
  return (
    <SectionPage>
      <Specimen title="Toggle series with the pointer or keyboard">
        <ChartLegend
          items={[
            { key: "booked", label: "Booked", slot: 1, shape: "line" },
            { key: "available", label: "Available", slot: 2, shape: "line" },
          ]}
          hidden={hidden}
          onToggle={(key) =>
            setHidden((current) => {
              const next = new Set(current);
              if (next.has(key)) next.delete(key);
              else next.add(key);
              return next;
            })
          }
        />
      </Specimen>
    </SectionPage>
  );
}
