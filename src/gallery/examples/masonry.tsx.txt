import { useState } from "react";
import { Button } from "glass-ux/button";
import { Image } from "glass-ux/image";
import { Lightbox, type LightboxImage } from "glass-ux/lightbox";
import { Masonry } from "glass-ux/masonry";
import { Surface } from "glass-ux/surface";
import { SCENE_IMAGES } from "glass-ux/gallery/fixtures/scenes";
import { ContainerLab, SectionPage, Specimen } from "glass-ux/gallery/kit";

const SCENES = ["harbour", "dunes", "ridge", "dusk"] as const;
const RATIOS = [1, 1.5, 0.75, 2, 1.25, 0.6, 1.5, 0.8, 1, 1.78, 0.67, 1.33];

interface Photo extends LightboxImage {
  ratio: number;
}

/** A fixed, fictional set: the four sample scenes in turn, each cropped to its own shape. */
const photo = (index: number): Photo => {
  const scene = SCENES[index % SCENES.length]!;
  return {
    id: `photo-${index + 1}`,
    src: SCENE_IMAGES[scene],
    alt: `Sample ${scene} view, number ${index + 1}`,
    caption: `View ${index + 1}`,
    width: 1600,
    height: 1200,
    ratio: RATIOS[index % RATIOS.length]!,
  };
};
const PHOTOS = Array.from({ length: 30 }, (_, index) => photo(index));
const NOTES = [
  "Booked for the quarterly review on the 14th.",
  "Two rooms free in the afternoon. Catering can be added up to a day before, and numbers can change until noon the day before the event.",
  "Open from seven.",
  "The loading dock is behind the east entrance. Deliveries are accepted between nine and three, and a booking is needed for anything over two pallets.",
  "Parking is limited to twelve spaces.",
  "Cancellations are free until seven days before the date.",
  "Accessible entrance on the north side.",
  "Projector and two screens available.",
];

function Photographs({
  initial = 18,
  adding = true,
  testId = "masonry-photos",
  ...rest
}: {
  initial?: number;
  adding?: boolean;
  testId?: string;
  dir?: "rtl";
}) {
  const [count, setCount] = useState(initial);
  const [open, setOpen] = useState(false);
  const [index, setIndex] = useState(0);
  const shown = PHOTOS.slice(0, count);
  return (
    <div className="flex w-full flex-col gap-3" data-testid={testId} {...rest}>
      {adding ? (
        <div>
          <Button
            size="sm"
            disabledReason={count >= PHOTOS.length ? "All photographs are shown" : undefined}
            onClick={() => setCount((current) => Math.min(PHOTOS.length, current + 6))}
          >
            Add 6 photographs
          </Button>
        </div>
      ) : null}
      <Masonry
        aria-label="Photographs"
        items={shown}
        getKey={(item) => item.id ?? item.src}
        getRatio={(item) => item.ratio}
        columnWidth="sm"
        renderItem={(item, position) => (
          <button
            type="button"
            className="fa-pressable block h-full w-full rounded-[var(--radius-media)]"
            aria-label={`Open ${item.caption}`}
            onClick={() => {
              setIndex(position);
              setOpen(true);
            }}
          >
            <Image src={item.src} alt="" ratio={item.ratio} hover="zoom" />
          </button>
        )}
      />
      <Lightbox
        images={shown}
        index={index}
        onIndexChange={setIndex}
        open={open}
        onOpenChange={setOpen}
      />
    </div>
  );
}

export default function MasonrySection() {
  return (
    <SectionPage>
      <Specimen
        title="Photographs"
        description="Each photograph's shape is known, so its box is reserved before it loads. Tab follows the way the grid reads, and a photograph opens in the Lightbox."
      >
        <Photographs />
      </Specimen>

      <Specimen
        title="Measured items"
        description="Items without a ratio are measured once they render; later items in their column move when the height settles."
      >
        <div className="w-full" data-testid="masonry-notes">
          <Masonry
            aria-label="Notes"
            items={NOTES}
            getKey={(note) => note}
            columnWidth="xs"
            renderItem={(note) => (
              <Surface material="card" radius="tile" padding="tile">
                <p className="type-body">{note}</p>
              </Surface>
            )}
          />
        </div>
      </Specimen>

      <Specimen
        title="Right to left"
        description="The first column is at the right and the order of reading follows it."
      >
        <Photographs initial={9} adding={false} testId="masonry-rtl" dir="rtl" />
      </Specimen>

      <ContainerLab>
        <div className="h-full overflow-auto p-2">
          <Masonry
            aria-label="Photographs"
            items={PHOTOS.slice(0, 12)}
            getKey={(item) => item.id ?? item.src}
            getRatio={(item) => item.ratio}
            columnWidth="xs"
            renderItem={(item) => <Image src={item.src} alt="" ratio={item.ratio} />}
          />
        </div>
      </ContainerLab>
    </SectionPage>
  );
}
