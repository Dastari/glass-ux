// Counters live outside React, so a remount shows as a second mount and not as a reset.
const COUNTS = new Map<
  string,
  {
    mounts: number;
    setups: number;
    cleanups: number;
  }
>();
const counts = (key: string) => {
  const entry = COUNTS.get(key) ?? { mounts: 0, setups: 0, cleanups: 0 };
  COUNTS.set(key, entry);
  return entry;
};

/** Mounts, effect setups and cleanups of the panel around it, written without React state. */
function Lifecycle({ name, label }: { name: string; label: string }) {
  const output = useRef<HTMLParagraphElement>(null);
  useState(() => {
    counts(name).mounts += 1;
    return null;
  });
  useEffect(() => {
    const entry = counts(name);
    const write = () => {
      if (output.current)
        output.current.textContent = `${label}: mounted ${entry.mounts}, effects started ${entry.setups}, stopped ${entry.cleanups}`;
    };
    entry.setups += 1;
    write();
    return () => {
      entry.cleanups += 1;
    };
  }, [name, label]);
  return (
    <p ref={output} className="type-caption text-fg-muted" data-testid={`pane-count-${name}`} />
  );
}

const ROWS = Array.from({ length: 40 }, (_, index) => `Survey point ${index + 1}`);

/** A panel with the state a remount would lose: an uncontrolled input, a scroll position and focus. */
function Report({ id }: { id: string }) {
  const { move } = usePaneLayout();
  return (
    <PanelBody>
      <div className="flex flex-col gap-3">
        <Lifecycle name={`workbench-${id}`} label="Report" />
        <Input aria-label="Report title" defaultValue="" placeholder="Title" />
        <Button
          size="sm"
          variant="secondary"
          className="self-start"
          onClick={() => move(id, { stack: "notes" })}
        >
          Move beside Notes
        </Button>
        <ul className="type-body" data-testid="pane-report-rows" tabIndex={0}>
          {ROWS.map((label) => (
            <li key={label}>{label}</li>
          ))}
        </ul>
      </div>
    </PanelBody>
  );
}

function Plain({ id, name, text }: { id: string; name: string; text: string }) {

}

  row(

    sized(column({ ...stack("notes", "history"), id: "notes" }, stack("session")), WIDE_TRACK),
  ),
);

function Workbench() {
  const [change, setChange] = useState("No change yet");
  const commits = useRef<HTMLDivElement>(null);
  const commitCount = useRef(0);
  return (
    <div className="flex w-full flex-col gap-2" data-testid="pane-workbench">
      <div ref={commits} className="h-120 w-full" data-commits="0">
        <Profiler
          id="workbench"
          onRender={() => {
            commitCount.current += 1;
            commits.current?.setAttribute("data-commits", String(commitCount.current));
          }}
        >
          <PaneLayout
            panel={workbenchPanel}
            defaultLayout={WORKBENCH_LAYOUT}
            storageKey="glass-ux.gallery.pane-layout"
            onLayoutChange={(_layout, { reason }) => setChange(`Last change: ${reason}`)}
          />
        </Profiler>
      </div>
      <p className="type-caption text-fg-muted" data-testid="pane-change">
        {change}
      </p>
    </div>
  );
}

  );
}
