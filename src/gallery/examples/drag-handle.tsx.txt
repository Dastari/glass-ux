import { useState } from "react";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { Button } from "glass-ux/button";
import { DragDropProvider, DragHandle, Draggable, DropZone, moveItem } from "glass-ux/drag-drop";

function Handles() {
  const [rows, setRows] = useState(["Front gallery", "Print room", "Courtyard", "Loading bay"]);
  const [opened, setOpened] = useState<string | null>(null);
  return (
    <DragDropProvider>
      <div className="flex w-full max-w-md flex-col gap-2">
        <DropZone
          id="rooms"
          label="Rooms"
          sort="vertical"
          render={<ul />}
          onDrop={(event) =>
            setRows((current) =>
              moveItem(current, event.source.index ?? 0, event.target.index ?? 0),
            )
          }
          className="flex flex-col gap-1"
        >
          {rows.map((room) => (
            <Draggable
              key={room}
              id={`room-${room}`}
              label={room}
              render={<li />}
              className="flex items-center gap-2 rounded-[var(--radius-item)] border border-[var(--line)] bg-[var(--glass-fill-control)] py-1 pe-1 ps-1 text-sm"
            >
              <DragHandle />
              <span className="min-w-0 flex-1 truncate">{room}</span>
              <Button size="sm" variant="ghost" onClick={() => setOpened(room)}>
                Open
              </Button>
            </Draggable>
          ))}
        </DropZone>
        <p className="h-5 text-xs text-[var(--fg-muted)]" aria-live="polite">
          {opened ? `Opened ${opened}` : ""}
        </p>
      </div>
    </DragDropProvider>
  );
}

export default function DragDropSection() {
  return (
    <SectionPage>
      <Specimen
        title="Handles"
        description="A DragHandle makes the grip the only place to start a drag, so the row's buttons keep working. On touch the grip drags at once; whole-item handles lift after a short press so lists still scroll."
        layout="stack"
      >
        <Handles />
      </Specimen>
    </SectionPage>
  );
}
