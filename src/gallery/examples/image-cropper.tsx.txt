import { useState } from "react";
import { Field } from "glass-ux/field";
import { FormStack } from "glass-ux/form";
import {
  ImageCropper,
  ImageInput,
  type ImageCropAspect,
  type ImageCropRect,
} from "glass-ux/image-cropper";
import { SCENE_IMAGES, WIDE_MARK } from "glass-ux/gallery/fixtures/scenes";
import { ContainerLab, SectionPage, Specimen } from "glass-ux/gallery/kit";

const SCENE = SCENE_IMAGES.harbour;
const SCENE_SIZE = { width: 1600, height: 1200 };
const ASPECTS = [
  { value: "free", label: "Free" },
  { value: 1, label: "Square" },
  { value: 16 / 9, label: "16:9" },
  { value: 3 / 4, label: "3:4" },
] as const;

function Report({ crop }: { crop: ImageCropRect | null }) {
  return (
    <output className="type-caption text-fg-muted" data-numeric data-testid="crop-report">
      {crop ? `${crop.width} × ${crop.height} at ${crop.x}, ${crop.y}` : "No crop"}
    </output>
  );
}

function LockedDemo() {
  const [crop, setCrop] = useState<ImageCropRect | null>(null);
  return (
    <div className="flex w-full flex-col items-center gap-2" data-testid="crop-locked">
      <ImageCropper
        className="w-full"
        src={SCENE}
        image={SCENE_SIZE}
        aspect={16 / 9}
        defaultValue={{ x: 320, y: 330, width: 960, height: 540 }}
        onValueChange={setCrop}
        minWidth={320}
      />
      <Report crop={crop} />
    </div>
  );
}

function FreeDemo() {
  const [crop, setCrop] = useState<ImageCropRect | null>(null);
  const [aspect, setAspect] = useState<ImageCropAspect>("free");
  return (
    <div className="flex w-full flex-col items-center gap-2" data-testid="crop-free">
      <ImageCropper
        className="w-full"
        src={SCENE}
        image={SCENE_SIZE}
        aspect={aspect}
        onAspectChange={setAspect}
        aspects={ASPECTS}
        defaultValue={{ x: 400, y: 300, width: 800, height: 600 }}
        onValueChange={setCrop}
        minWidth={120}
        minHeight={120}
      />
      <Report crop={crop} />
    </div>
  );
}

export default function ImageCropperSection() {
  return (
    <SectionPage>
      <Specimen
        title="Locked aspect"
        description="The image moves under a frame of one shape: drag, pinch, wheel or the arrow keys, and the slider zooms. The crop is reported in source pixels once a gesture ends."
      >
        <LockedDemo />
      </Specimen>

      <Specimen
        title="Free and preset shapes"
        description="Free shows the whole image with a box to move and resize. The host supplies the presets. The number fields are the way to place the crop without dragging."
      >
        <FreeDemo />
      </Specimen>

      <Specimen
        title="Round"
        description="A round crop is still a rectangle of the image; the corners are dimmed."
      >
        <ImageCropper
          className="w-full"
          src={SCENE}
          image={SCENE_SIZE}
          aspect={1}
          shape="round"
          fields={false}
        />
      </Specimen>

      <Specimen title="States" layout="grid">
        <FormStack>
          <Field label="Empty" nativeLabel={false}>
            <ImageInput aspect={16 / 9} />
          </Field>
          <Field label="Chosen" nativeLabel={false}>
            <ImageInput aspect={16 / 9} defaultValue={SCENE} alt="Fictional harbour" />
          </Field>
          <Field label="Missing" nativeLabel={false} error="Add an image">
            <ImageInput aspect={16 / 9} invalid />
          </Field>
          <Field label="A wide logo, free aspect" nativeLabel={false}>
            <div data-testid="image-input-wide">
              <ImageInput defaultValue={WIDE_MARK} alt="Fictional wordmark" />
            </div>
          </Field>
        </FormStack>
        <FormStack>
          <Field label="Read-only" nativeLabel={false}>
            <ImageInput aspect={1} defaultValue={SCENE} readOnly alt="Fictional harbour" />
          </Field>
          <Field label="Disabled" nativeLabel={false}>
            <ImageInput
              aspect={1}
              shape="round"
              defaultValue={SCENE}
              disabled
              alt="Fictional harbour"
            />
          </Field>
          <ImageCropper aspect={16 / 9} fields={false} disabled />
        </FormStack>
      </Specimen>

      <ContainerLab>
        <div className="h-full overflow-auto p-1">
          <ImageCropper src={SCENE} image={SCENE_SIZE} aspect="free" />
        </div>
      </ContainerLab>
    </SectionPage>
  );
}
