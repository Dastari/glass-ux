  IconAccessPoint,
  IconDeskPhone,

  IconNetworkSwitch,

} from "glass-ux/icons";

import {
  TopologyCanvas,
  TopologySkeleton,
  TopologyViewToggle,
  type TopologyClusterSpec,
  type TopologyNode,
  type TopologyNodeMenu,
  type TopologyView,
} from "glass-ux/topology";

/** A node's kind in the specimens: the glyph and its name. */
type Kind = "router" | "switch" | "ap" | "computer" | "server" | "phone" | "printer";

interface Thing {
  /** Semantic entity or content kind, used to choose labels and rendering. */
  kind: Kind;
  /** Controlled view state, including loading, empty and error states. */
  state: EntityState;
}

const ICONS = {
  router: IconRouter,
  switch: IconNetworkSwitch,
  ap: IconAccessPoint,
  computer: IconEndpoint,
  server: IconServer,
  phone: IconDeskPhone,
  printer: IconPrinter,
};
const LABELS: Record<Kind, string> = {
  router: "Router",
  switch: "Switch",
  ap: "Access point",
  computer: "Computer",
  server: "Server",
  phone: "Phone",
  printer: "Printer",
};
const ONLINE: EntityState = { id: "online", label: "Online", tone: "success" };
const OFFLINE: EntityState = { id: "offline", label: "Offline", tone: "neutral" };
const IN_CALL: EntityState = { id: "in-call", label: "In a call", tone: "info", activity: true };
const CLUSTERS: Record<string, TopologyClusterSpec> = {
  servers: { label: "Servers", order: 1 },
  computers: { label: "Computers", order: 2 },
  phones: { label: "Phones", order: 3 },
  printers: { label: "Printers", order: 4 },
};
const at = (
  id: string,
  parentId: string | null,
  kind: Kind,
  extra: Partial<TopologyNode<Thing>> = {},
  state = ONLINE,
): TopologyNode<Thing> => ({
  id,
  parentId,
  label: id,
  data: { kind, state },
  ...extra,
});

/** About 300 nodes: a school with six switches, two access points each and their clients. */
function school(): TopologyNode<Thing>[] {
  const nodes: TopologyNode<Thing>[] = [at("HVS-GW", null, "router")];
  for (let s = 1; s <= 6; s++) {
    nodes.push(at(`HVS-SW${s}`, "HVS-GW", "switch", { link: { kind: "wired", speedMbps: 10000 } }));
    for (let a = 1; a <= 2; a++) {
      nodes.push(
        at(`HVS-AP${s}${a}`, `HVS-SW${s}`, "ap", { link: { kind: "wired", speedMbps: 2500 } }),
      );
      for (let c = 1; c <= 12; c++)
        nodes.push(
          at(
            `HVS-L${s}${a}-${c}`,
            `HVS-AP${s}${a}`,
            "computer",
            { cluster: "computers", link: { kind: "wireless" } },
            c % 7 ? ONLINE : OFFLINE,
          ),
        );
    }
    for (let c = 1; c <= 21; c++)
      nodes.push(
        at(`HVS-D${s}-${c}`, `HVS-SW${s}`, c % 5 ? "computer" : "phone", {
          cluster: c % 5 ? "computers" : "phones",
          link: { kind: "wired", speedMbps: c % 4 ? 1000 : 100 },
        }),
      );
  }
  return nodes;
}

const ACTIONS = defineActions<
  Thing & {
    name: string;
  }
>("device", [
  { id: "open", label: "Open", group: "open", run: () => undefined },
  { id: "ping", label: "Ping", group: "connect", run: () => undefined },
  { id: "settings", label: "Settings…", group: "manage", run: () => undefined },
]);

function Tile({ node }: { node: TopologyNode<Thing> }) {
  const Icon = ICONS[node.data.kind];
  return (
    <EntityIdentity
      tier="icon"
      name={node.label}
      kind={LABELS[node.data.kind]}
      icon={<Icon />}
      state={node.data.state}
    />
  );
}

function DemoMap({
  nodes,
  minimap = true,
  listable = false,
}: {
  nodes: TopologyNode<Thing>[] | undefined;
  minimap?: boolean;
  listable?: boolean;
}) {
  const [selected, setSelected] = useState<string | null>(null);
  const [view, setView] = useState<TopologyView>("map");
  const renderNode = useCallback((node: TopologyNode<Thing>) => <Tile node={node} />, []);
  const describe = useCallback(
    (node: TopologyNode<Thing>) =>
      `${LABELS[node.data.kind]} ${node.label}, ${node.data.state.label}`,
    [],
  );
  const menu = useCallback(
    (node: TopologyNode<Thing>): TopologyNodeMenu => ({
      set: ACTIONS,
      targets: [{ ...node.data, name: node.label }],
      label: node.label,
    }),
    [],
  );
  return (
    <TopologyCanvas<Thing>
      label="Network map"
      nodes={nodes}
      clusters={CLUSTERS}
      renderNode={renderNode}
      describeNode={describe}
      selectedId={selected}
      onSelect={setSelected}
      nodeMenu={menu}
      minimap={minimap}
      view={view}
      onViewChange={listable ? setView : undefined}
      list={
        listable ? (
          <div className="flex flex-col items-start gap-3">
            <TopologyViewToggle view="list" onViewChange={setView} size="sm" />
            <p className="type-body text-fg-muted">
              The caller's list goes here: a DataTable of the same nodes.
            </p>
          </div>
        ) : undefined
      }
      emptyText="No devices seen on this network yet"
    />
  );
}

export default function TopologySection() {
  const big = useMemo(() => school(), []);
  return (
    <SectionPage>
      <Specimen
        title="Loading and empty"
        description="The skeleton fills the map's own box while the layout loads (it is code-split), so nothing moves when the map lands. An empty map says so in the same box."
      >
        <SwapSlot
          id="topology"
          className="h-90 w-160 max-w-full"
          skeleton={
            <div className="fa-topology">
              <TopologySkeleton />
            </div>
          }
          loaded={<DemoMap nodes={OFFICE.slice(0, 6)} />}
        />
        <div className="h-90 w-120 max-w-full">
          <DemoMap nodes={[]} />
        </div>
      </Specimen>
    </SectionPage>
  );
}
