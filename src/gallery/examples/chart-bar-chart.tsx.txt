import type { ComponentProps } from "react";
import { BarChart, type ChartState } from "glass-ux/charts";
import {
  ContainerLab,
  SectionPage,
  Specimen,
  TierBox,
  TierRow,
} from "glass-ux/gallery/kit";
import { SkeletonSwaps } from "glass-ux/gallery/sections/charts.kit";

const sample = {
  categories: ["Planning", "Delivery", "Review"],
  series: [{ key: "hours", label: "Hours", data: [32, 48, 20] }],
  unit: "h",
} satisfies Omit<ComponentProps<typeof BarChart>, "label">;
const render = (state: ChartState = "ready") => (
  <BarChart
    {...sample}
    label="BarChart · fictional studio projects"
    title="Studio projects"
    state={state}
    emptyText="No project measurements yet"
    errorText="Measurements could not load"
    onRetry={() => undefined}
  />
);

export default function BarChartSection() {
  return (
    <SectionPage>
      <Specimen
        title="BarChart"
        description="A focused example using fictional studio project measurements. Hover or use the keyboard to inspect values; the chart's table preserves the exact data."
        wide
      >
        <div className="charts-pane" data-testid="chart-bar-chart">
          {render()}
        </div>
      </Specimen>
      <Specimen
        title="Responsive sizes"
        description="The same BarChart adapts to row, tile and full-panel sizes."
        wide
      >
        <TierRow>{() => render()}</TierRow>
      </Specimen>
      <Specimen
        title="Loading, empty and error"
        description="States stay in the chart's own box. Swap the skeleton to compare its dimensions."
        wide
      >
        <SkeletonSwaps id="bar-chart" render={render} />
        <div className="gallery-tiers">
          <TierBox size="compact" label="Empty">
            {render("empty")}
          </TierBox>
          <TierBox size="full" label="Error">
            {render("error")}
          </TierBox>
        </div>
      </Specimen>
      <ContainerLab>{() => render()}</ContainerLab>
    </SectionPage>
  );
}
