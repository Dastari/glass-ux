import { z } from "zod";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { DEMO_MAP_CONFIG } from "glass-ux/gallery/fixtures/map";
import { coordinatesSchema, type LatLng } from "glass-ux/coordinates-field";
import { Form, FormActions, FormStack, formBehaviour, useAppForm } from "glass-ux/form";
import { wait } from "glass-ux/gallery/sections/forms.fixtures";

const TASMANIA = [143.8, -43.7, 148.5, -39.5] as const;

function BoundDemo() {
  const form = useAppForm({
    ...formBehaviour,
    defaultValues: {
      position: { lat: -42.88334, lng: 147.32688 } as LatLng | null,
      antenna: null as LatLng | null,
    },
    validators: {
      onDynamic: z.object({
        position: coordinatesSchema({ required: true }),
        antenna: coordinatesSchema(),
      }),
    },
    onSubmit: () => wait(600),
  });
  return (
    <Form form={form} aria-label="Positions" className="w-full" data-testid="coordinates-form">
      <FormStack>
        <form.AppField name="position">
          {(field) => (
            <field.CoordinatesField
              label="Site position"
              map={DEMO_MAP_CONFIG}
              mapFallback={TASMANIA}
            />
          )}
        </form.AppField>
        <form.AppField name="antenna">
          {(field) => <field.CoordinatesField label="Antenna" optional locate={false} />}
        </form.AppField>
        <FormActions>
          <form.SubmitButton>Save</form.SubmitButton>
        </FormActions>
      </FormStack>
    </Form>
  );
}

export default function CoordinatesFieldSection() {
  return (
    <SectionPage>
      <Specimen
        title="Pick on map and use my location"
        description="The map pin opens a picker: click the map or drag the pin. The location button asks the browser for the device's position; a site whose location is blocked says how to allow it."
      >
        <BoundDemo />
      </Specimen>
    </SectionPage>
  );
}
