import { useRef, useState, type FormEvent } from "react";
import { Button } from "glass-ux/button";
import { Field } from "glass-ux/field";
import { Input } from "glass-ux/input";
import {
  KeyboardTarget,
  KeyboardToggle,
  OnScreenKeyboardProvider,
  useOnScreenKeyboard,
} from "glass-ux/on-screen-keyboard";
import { Surface } from "glass-ux/surface";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";

function KioskForm() {
  const keyboard = useOnScreenKeyboard();
  const [signedIn, setSignedIn] = useState<string | null>(null);
  const submit = (event: FormEvent<HTMLFormElement>) => {
    event.preventDefault();
    const name = new FormData(event.currentTarget).get("name");
    keyboard.close();
    setSignedIn(typeof name === "string" && name.trim() ? name.trim() : "Visitor");
  };
  return (
    <div className="gallery-osk-frame-scroll" data-scroll-container="">
      <div className="gallery-osk-frame-header">
        <h3 className="type-title">Visitor sign-in</h3>
        <KeyboardToggle />
      </div>
      <KeyboardTarget>
        <form className="gallery-osk-frame-form" aria-label="Visitor sign-in" onSubmit={submit}>
          <Field label="Name">
            <Input name="name" autoComplete="off" enterKeyHint="next" />
          </Field>
          <Field label="Email">
            <Input name="email" type="email" autoComplete="off" enterKeyHint="next" />
          </Field>
          <Field label="Mobile">
            <Input name="mobile" type="tel" autoComplete="off" />
          </Field>
          <Field label="Visiting">
            <Input name="host" autoComplete="off" enterKeyHint="done" />
          </Field>
          <div className="gallery-osk-toolbar">
            <Button type="submit" variant="primary">
              Sign in
            </Button>
          </div>
          <p
            className="gallery-osk-status"
            data-tone={signedIn ? "success" : undefined}
            data-align="start"
            aria-live="polite"
          >
            {signedIn ? `Signed in: ${signedIn}` : " "}
          </p>
        </form>
      </KeyboardTarget>
    </div>
  );
}

function KioskDemo() {
  const frame = useRef<HTMLDivElement>(null);
  return (
    <Surface ref={frame} material="panel" radius="surface" className="gallery-osk-frame">
      <OnScreenKeyboardProvider show="focus" container={frame}>
        <KioskForm />
      </OnScreenKeyboardProvider>
    </Surface>
  );
}

export default function OnScreenKeyboardSection() {
  return (
    <SectionPage>
      <Specimen
        title="Kiosk sign-in"
        description='OnScreenKeyboardProvider with show="focus" opens the drawer when a field inside KeyboardTarget gains focus, with a layout for each input type. The drawer docks to this frame (container) instead of the viewport.'
      >
        <KioskDemo />
      </Specimen>
    </SectionPage>
  );
}
