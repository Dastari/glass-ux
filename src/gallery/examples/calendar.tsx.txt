import { useState } from "react";
import { Example, SectionPage, Specimen } from "glass-ux/gallery/kit";
import {
  Calendar,
  addDays,
  formatTimeText,
  timeSlots,
  todayIso,
  type DateRange,
  type IsoDate,
  type TimeValue,
} from "glass-ux/date-picker";
import { TimeList } from "glass-ux/ui/date-picker/time-list";

const today = todayIso();

function SingleCalendar() {
  const [value, setValue] = useState<IsoDate | null>(addDays(today, 3));
  return (
    <Example label={value ?? "Nothing picked"}>
      <div
        className="glass-overlay fa-popup fa-popover fa-popover-preview fa-date-popover"
        data-size="sm"
      >
        <div className="fa-popover-body">
          <Calendar value={value} onValueChange={setValue} aria-label="Maintenance day" />
        </div>
      </div>
    </Example>
  );
}

function RangeCalendar() {
  const [value, setValue] = useState<DateRange | null>({
    start: addDays(today, -9),
    end: addDays(today, -2),
  });
  return (
    <Example label={value ? `${value.start} – ${value.end}` : "Nothing picked"}>
      <div
        className="glass-overlay fa-popup fa-popover fa-popover-preview fa-date-popover"
        data-size="sm"
      >
        <div className="fa-popover-body">
          <Calendar
            mode="range"
            value={value}
            onValueChange={setValue}
            aria-label="Reporting period"
          />
        </div>
      </div>
    </Example>
  );
}

function DateTimePickerPreview() {
  const [day, setDay] = useState<IsoDate>(addDays(today, 2));
  const [time, setTime] = useState<TimeValue>("09:30");
  const times = timeSlots({ step: 30, min: "07:00", max: "19:00" });
  return (
    <Example label={`${day}T${time}`}>
      <div
        className="glass-overlay fa-popup fa-popover fa-popover-preview fa-date-popover fa-date-time-popover"
        data-size="sm"
      >
        <div className="fa-popover-body">
          <div className="fa-date-time-picker">
            <Calendar value={day} onValueChange={setDay} aria-label="Change window day" />
            <div className="fa-date-time-times">
              <TimeList
                times={times}
                value={time}
                format={(next) => formatTimeText(next)}
                onPick={setTime}
                aria-label="Change window time"
              />
            </div>
          </div>
        </div>
      </div>
    </Example>
  );
}

export default function DatePickerSection() {
  return (
    <SectionPage>
      <Specimen
        title="Calendar"
        description="Arrow keys move by day and week, Home and End to the week's ends, Page Up and Page Down by month (Shift: year), Enter picks. In a range, the first pick is the start; the band follows the pointer until the second."
      >
        <SingleCalendar />
        <RangeCalendar />
        <Example label="Bounded: the next 30 days">
          <div
            className="glass-overlay fa-popup fa-popover fa-popover-preview fa-date-popover"
            data-size="sm"
          >
            <div className="fa-popover-body">
              <Calendar min={today} max={addDays(today, 30)} aria-label="Next 30 days" />
            </div>
          </div>
        </Example>
      </Specimen>

      <Specimen
        title="Calendar and times"
        description="The date-and-time popup: the calendar beside the times, which wrap below it in a narrow popup. Picking a time closes it."
      >
        <DateTimePickerPreview />
      </Specimen>
    </SectionPage>
  );
}
