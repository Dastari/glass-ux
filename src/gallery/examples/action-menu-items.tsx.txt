import { ActionMenuItems, defineActions, resolveActions } from "glass-ux/actions";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { toast } from "glass-ux/toast";
import { Menu } from "glass-ux/menu";
import { Button } from "glass-ux/button";

export default function ActionMenuItemsExample() {
  const actions = defineActions<{
    id: string;
  }>("exhibitions", [
    {
      id: "open",
      label: "Open exhibition",
      group: "open",
      run: () => toast.info("Sample exhibition opened"),
    },
  ]);
  const targets = [{ id: "meridian" }];
  const groups = resolveActions(actions, targets, "menu");
  return (
    <SectionPage>
      <Specimen title="Rows from shared action definitions">
        <Menu.Root>
          <Menu.Trigger render={<Button>Exhibition actions</Button>} />
          <Menu.Popup>
            <ActionMenuItems groups={groups} targets={targets} surface="menu" />
          </Menu.Popup>
        </Menu.Root>
      </Specimen>
    </SectionPage>
  );
}
