import { useState, type ReactNode } from "react";
import { LocaleProvider, useLocale } from "glass-ux/locale";
import { DateField, TimeInput } from "glass-ux/date-picker";
import { Field } from "glass-ux/field";
import { FormStack } from "glass-ux/form";
import { MoneyInput } from "glass-ux/money-field";
import { NumberField } from "glass-ux/number-field";
import { Segmented } from "glass-ux/segmented";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";

const ISLAND_LOCALES = [
  { value: "en-US", label: "en-US" },
  { value: "en-GB", label: "en-GB" },
  { value: "de-DE", label: "de-DE" },
];

/** The same stored values, formatted by whichever locale encloses them. */
function Values({ heading }: { heading: ReactNode }) {
  const locale = useLocale();
  return (
    <div className="flex min-w-0 flex-col gap-3" data-locale-sample={locale}>
      <div className="flex min-h-8 flex-wrap items-center justify-between gap-2">{heading}</div>
      <FormStack>
        <Field label="Date">
          <DateField defaultValue="2026-10-03" />
        </Field>
        <Field label="Time">
          <TimeInput defaultValue="21:05" />
        </Field>
        <Field label="Amount">
          <MoneyInput defaultValue={123450} />
        </Field>
        <Field label="Count">
          <NumberField defaultValue={1284.5} format={{ maximumFractionDigits: 1 }} />
        </Field>
      </FormStack>
    </div>
  );
}

export default function LocaleSection() {
  const [island, setIsland] = useState("en-US");
  return (
    <SectionPage>
      <Specimen
        title="A LocaleProvider region"
        description={
          <>
            A component&apos;s <code>locale</code> prop wins, then the nearest{" "}
            <code>LocaleProvider</code>, then <code>configureLocale()</code>, then the page&apos;s{" "}
            <code>&lt;html lang&gt;</code>, then en-US.
          </>
        }
        layout="grid"
      >
        <Values
          heading={
            <span className="type-label">
              Page <code>&lt;html lang&gt;</code>
            </span>
          }
        />
        <LocaleProvider locale={island}>
          <Values
            heading={
              <>
                <span className="type-label">LocaleProvider</span>
                <Segmented
                  size="sm"
                  options={ISLAND_LOCALES}
                  value={island}
                  onValueChange={setIsland}
                  aria-label="Provider locale"
                />
              </>
            }
          />
        </LocaleProvider>
      </Specimen>
    </SectionPage>
  );
}
