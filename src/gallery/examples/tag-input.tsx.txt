import { useState } from "react";

import { ContainerLab, SectionPage, Specimen } from "glass-ux/gallery/kit";

import { IconGlobe, IconMail, IconTag } from "glass-ux/icons";

import { Combobox } from "glass-ux/combobox";

import { Field } from "glass-ux/field";

import { FormStack } from "glass-ux/form";

import { MultiSelect } from "glass-ux/multi-select";

import { TagInput } from "glass-ux/tag-input";

import { LABEL_SUGGESTIONS, ORGANISATIONS, TAGS } from "glass-ux/gallery/sections/forms.fixtures";

const EMAIL = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;

const validateEmail = (value: string) => (EMAIL.test(value) ? null : "Enter an email such as name@example.com.au");

const IPV4 = /^(25[0-5]|2[0-4]\d|1\d\d|[1-9]?\d)(\.(25[0-5]|2[0-4]\d|1\d\d|[1-9]?\d)){3}$/;

const HOSTNAME = /^(?=.{1,253}$)([a-z0-9](-?[a-z0-9])*)(\.[a-z0-9](-?[a-z0-9])*)*$/i;

const validateHost = (value: string) => (IPV4.test(value) || HOSTNAME.test(value) ? null : "Not an IPv4 address or hostname");

/** A TagInput in a Field, with the first chip problem in the message line. */
function Recipients() {
    const [error, setError] = useState<string>();
    return (<Field label="Send the report to" hint="Paste a list — commas, semicolons or new lines." error={error}>
      <TagInput icon={<IconMail />} defaultValue={["avery.quinn@orbit.example.com.au", "mira.rowan@northstar.example"]} normalize={(value) => value.toLowerCase()} validate={validateEmail} onErrorChange={setError} placeholder="name@example.com.au" max={10} aria-label="Send the report to"/>
    </Field>);
}

function Hosts() {
    const [error, setError] = useState<string>();
    return (<Field label="Allowed hosts" hint="IPv4 addresses or hostnames, one per line when pasting." error={error}>
      <TagInput icon={<IconGlobe />} validate={validateHost} onErrorChange={setError} normalize={(value) => value.toLowerCase()} placeholder="10.20.0.10 or files.northstar.local"/>
    </Field>);
}

export default function ComboboxSection() {
    return (<SectionPage>
      <></>

      <></>

      <></>

      <></>

      <></>

      <Specimen title="Chips you type" description="TagInput: Enter, comma, semicolon or Tab makes a chip; pasting a list adds them all. Values are tidied and de-duplicated — adding one again flashes the existing chip. Invalid chips stay, in danger, with the reason as their tooltip and in the message line. Enter or a double-click edits a chip.">
        <FormStack>
          <Recipients />
          <Hosts />
          <Field label="Labels" hint="Suggestions, or create your own.">
            <TagInput suggestions={LABEL_SUGGESTIONS} defaultValue={["Front counter"]} placeholder="Add a label" icon={<IconTag />}/>
          </Field>
          <Field label="Disabled" disabled>
            <TagInput defaultValue={["kiosk"]} disabled/>
          </Field>
        </FormStack>
      </Specimen>

      <></>

      <></>

      <ContainerLab>
        {(preset) => preset?.id === "row" ? (<div className="flex h-full items-center gap-2 px-1">
              <Combobox size="sm" options={ORGANISATIONS} defaultValue="cedarvale" aria-label="Organisation" className="w-64"/>
              <MultiSelect size="sm" options={TAGS} defaultValue={["production", "all-in-one", "front-counter", "finance"]} aria-label="Tags"/>
            </div>) : preset?.id === "header" ? (<div className="flex h-full items-center px-2">
              <TagInput wrap={false} defaultValue={["avery.quinn@orbit.example.com.au", "mira.rowan@northstar.example", "it@northstar.example"]} aria-label="Recipients"/>
            </div>) : (<div className="h-full overflow-y-auto" data-scroll-container>
              <FormStack className="p-3">
                <Field label="Organisation">
                  <Combobox options={ORGANISATIONS} defaultValue="cedarvale"/>
                </Field>
                <Field label="Tags">
                  <MultiSelect wrap options={TAGS} defaultValue={["production", "all-in-one", "front-counter", "finance"]}/>
                </Field>
                <Field label="Recipients">
                  <TagInput defaultValue={["avery.quinn@orbit.example.com.au", "it@northstar.example"]}/>
                </Field>
              </FormStack>
            </div>)}
      </ContainerLab>

      <></>

      <></>
    </SectionPage>);
}
