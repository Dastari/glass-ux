import { useState } from "react";
import { z } from "zod";
import { Form, FormActions, FormStack, formBehaviour, useAppForm } from "glass-ux/form";
import { imageSchema, type ImageFieldValue } from "glass-ux/image-cropper";
import { SectionPage, Specimen } from "glass-ux/gallery/kit";
import { wait } from "glass-ux/gallery/sections/forms.fixtures";

function FormDemo() {
  const [edits, setEdits] = useState(0);
  const form = useAppForm({
    ...formBehaviour,
    defaultValues: { cover: null as ImageFieldValue },
    validators: { onDynamic: z.object({ cover: imageSchema({ required: true }) }) },
    onSubmit: () => wait(600),
  });
  return (
    <Form form={form} aria-label="Album" className="w-full" data-testid="image-form">
      <FormStack>
        <form.AppField name="cover">
          {(field) => (
            <field.ImageField
              label="Cover"
              required
              aspect={1}
              onEdit={() => setEdits((count) => count + 1)}
            />
          )}
        </form.AppField>
        <output
          className="type-caption text-fg-muted"
          data-numeric
        >{`Host editor opened ${edits} times`}</output>
        <FormActions>
          <form.SubmitButton>Save album</form.SubmitButton>
        </FormActions>
      </FormStack>
    </Form>
  );
}

export default function ImageCropperSection() {
  return (
    <SectionPage>
      <Specimen
        title="In a form"
        description="ImageField holds a URL, a new crop or null. onEdit opens the host's own editor, and the field does not know what that is."
      >
        <FormDemo />
      </Specimen>
    </SectionPage>
  );
}
