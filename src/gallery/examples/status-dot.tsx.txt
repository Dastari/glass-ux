import { Example, SectionPage, Specimen } from "glass-ux/gallery/kit";
import { STATUS, StatusBadge, StatusDot, type StatusKind } from "glass-ux/badge";

const STATUSES = Object.keys(STATUS) as StatusKind[];

export default function BadgeSection() {
  return (
    <SectionPage>
      <Specimen
        title="Endpoint status"
        description="One definition per state, shared by badges, dots, the tree and the dock. Only live and connecting pulse; installing spins. Reduced motion stops the pulse."
        layout="stack"
      >
        <Example label="StatusBadge · md">
          {STATUSES.map((status) => (
            <StatusBadge key={status} status={status} />
          ))}
        </Example>
        <Example label="StatusBadge · sm">
          {STATUSES.map((status) => (
            <StatusBadge key={status} status={status} size="sm" />
          ))}
        </Example>
        <Example label="StatusDot">
          {STATUSES.map((status) => (
            <span
              key={status}
              className="inline-flex items-center gap-inline type-caption text-fg-muted"
            >
              <StatusDot status={status} />
              {STATUS[status].label}
            </span>
          ))}
        </Example>
      </Specimen>
    </SectionPage>
  );
}
