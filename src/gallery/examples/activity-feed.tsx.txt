function TicketTimeline() {
  const [items, setItems] = useState(() => ticketHistory(Date.now()));
  // Arrivals go to the end of an oldest-first timeline, so nothing above them moves (R17.3).
  const addCheck = () =>
    setItems((current) => [
      ...current,
      {
        id: `check-${current.length}`,
        content: (
          <>
            Monitoring reported <strong>HF-PRN-07</strong> online
          </>
        ),
        at: Date.now(),
        icon: <IconCheck />,
        tone: "success",
      },
    ]);
  return (
    <div className="flex w-full max-w-2xl flex-col gap-4" data-testid="activity-ticket">
      <div>
        <Button size="sm" onClick={addCheck}>
          Add a monitoring event
        </Button>
      </div>
      <ActivityFeed
        aria-label="Ticket history"
        items={items}
        variant="timeline"
        order="oldest"
        timePlacement="inline"
      />
    </div>
  );
}

/** The mark for an arrival: it takes no room, so setting and clearing it moves nothing. */
function NewEvents() {
  const [marked, setMarked] = useState(true);
  const [started] = useState(() => Date.now());
  const items: ActivityItem[] = [
    {
      id: "scan",
      content: "Dock 3 reported a label scan fault",
      at: started - 2 * MINUTE,
      icon: <IconFlag />,
      isNew: marked,
    },
    {
      id: "assigned",
      content: "Ari Moss assigned the fault to Sam Okafor",
      at: started - 9 * MINUTE,
      icon: <IconUserCheck />,
    },
    {
      id: "restart",
      content: "HF-PRN-07 restarted",
      at: started - 25 * MINUTE,
      icon: <IconActivity />,
    },
  ];
  return (
    <div className="flex w-full max-w-xl flex-col gap-4" data-testid="activity-new">
      <Switch checked={marked} onCheckedChange={setMarked} label="The first event is new" />
      <ActivityFeed aria-label="Dock events" items={items} />
      <ActivityFeed
        aria-label="Dock history"
        items={items}
        variant="timeline"
        timePlacement="inline"
      />
      <ActivityFeed aria-label="Dock events to select" items={items} defaultSelectedId="assigned" />
    </div>
  );
}
